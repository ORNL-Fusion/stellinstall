206  CALL allocate_silo

208 #endif

221  npmax=(iam+1)*ndum

244  ENDIF
245  c_flx(3) = r_cyl(2)

247  mpol_w, ntmax_w, lthreed_w, lasym_w, info, nfe, fmin) ! FMIN not used in what follows.
248  c_flx_old = c_flx

253  uu = c_flx(2)
254  vv = c_flx(3)/nfp_w
255  index1 = index1+1
256  CALL dump_pressure(vv, ss, uu, info, index1) ! Now, get quantities and dump them on file
257  ENDDO
258  ENDDO
259  ENDDO

264 !

266 
267  IF (iam .EQ. 0) THEN
268  WRITE(*,'(/,10x,a)') 'Completed.'
269  WRITE(*,'(10x,a)') 'Pressure (PGPLOT) using CYLINDRICAL coords..' ! Use uniformly-spaced grid in (R,v,Z)

273  nmpi = 0

275  ndum = maxprocs/min(nprocs, maxprocs)
276  npmin=iam*ndum+1
277  npmax=(iam+1)*ndum
278  IF (iam .EQ. min(nprocs-1,maxprocs-1)) npmax = maxprocs
279  ALLOCATE (pprof(ndum+(maxprocs-ndum*nprocs)))

281

286  DO jk = 1, nus

288  IF (nmpi .LT. npmin) cycle
289  IF (nmpi .GT. npmax) GOTO 199

291 !! CALL flx_to_cyl(ss, uu, vv, rr, zz)

294  ENDDO
295  ENDDO
296  ENDDO
297 
298  199 CONTINUE
299 !

301 !
302  CALL write_pressure_data(nunit_pres1, ndum, maxprocs)
303 
304  IF (iam .EQ. 0) THEN
305  WRITE(*,'(/,10x,a)') 'Completed.'
306  WRITE(*,'(10x,a)') 'Pressure (PGPLOT) using SIESTA coords..' ! Use uniformly-spaced grid in (s,u,v)
307  END IF
308

310 ! Calculation in uniformly-spaced grid for Tracing and SILO output if requested
311 !
312  tracing: IF (l_tracing .OR. l_silo_output) THEN ! Do 3D calculation only if L_TRACING = T or SILO_OUTPUT desired

314  WRITE(*,'(/,10x,a)') 'Creating additional files:'

363 !

365  DO lk = 1, nvs

367  lendpt = (jk.NE.nus .AND. lk.NE.nvs) ! Skip angle end points (0,2p]
368  lbfld = .false.
369  DO ik = 2, nss
370  ss = (ik-1)*oss ! Actual SIESTA S-coordinate
371  IF (lendpt .AND. l_tracing) THEN
372  nmpi = nmpi+1
373  lbfld = (nmpi.GE.npmin .AND. nmpi.LE.npmax)
374  END IF
375  IF (l_silo_output) THEN
376  nmpi_s = nmpi_s+1
377  lsilo = (nmpi_s.GE.npmin_s .AND. nmpi_s.LE.npmax_s)
378  END IF
379  IF (nmpi.GT.npmax .AND. nmpi_s.GT.npmax_s) GOTO 299

382  CALL flx_to_cyl(ss, uu, vv, rr, zz)
383  CALL dump_quantities(rr, zz, vv, ss, uu, lbfld, lsilo)
384  ENDDO
385  ENDDO
386  ENDDO
387 
388  299 CONTINUE

409  ENDIF
410 #endif

1127  SUBROUTINE toijsp_1p(XMN, SS, THETA, ZETA, XUV, DFLAG, &
1128  iparity, ihalf)
1129 !
1130 ! DESCRIPTION: calculates X on ONE POINT in space (SS,THETA,ZETA) from its Fourier harmonics X_MN.
1131 ! IPARITY = 0, means the quantity X (NOT any of its derivatives!) is COSINE (even); = 1, X is SINE (odd)
1132 ! IHALF= 0, means quantity is on the radial full mesh; =1, on the half radial mesh
1133 ! DFLAG= 0b001 NO/YES radial deriv. DFLAG= 0b010 => NO/YES THETA-Deriv. DFLAG= 0b100 => NO/YES ZETA-Deriv.
1134 !
1135  USE stel_kinds
1136  USE fourier, ONLY: orthonorm, b_ds, b_du, b_dv, b_sum
1137  IMPLICIT NONE
1138 !-----------------------------------------------
1139 ! D u m m y A r g u m e n t s
1140 !-----------------------------------------------
1141  INTEGER, INTENT(IN):: ihalf ! IHALF = 0, FULL (radial) MESH); = 1, HALF MESH
1142  REAL(rprec), DIMENSION(0:mpol,-ntor:ntor,ns), & !New Style: ns LAST ARG on input
1143  INTENT(IN):: xmn
1144  INTEGER, INTENT(IN):: dflag
1145  REAL(rprec), INTENT(IN):: ss, theta, zeta
1146  REAL(rprec), INTENT(INOUT):: xuv
1147  INTEGER, INTENT(IN):: iparity ! IPARITY = 0, cosine (EVEN); = 1, sine (ODD)
1148 !-----------------------------------------------
1149 ! L o c a l V a r i a b l e s
1150 !-----------------------------------------------
1151  INTEGER :: m, n
1152  INTEGER :: ns_in, ns_out, iout, ndum, istat
1153  REAL(rprec), ALLOCATABLE, DIMENSION(:,:,:):: work0
1154  REAL(rprec), ALLOCATABLE, DIMENSION(:,:):: work1, work2
1155  REAL(rprec), ALLOCATABLE, DIMENSION(:):: xuv_i
1156  REAL(rprec) :: arg, d1, d2, ss_in, aux1, aux2
1157  REAL(rprec):: lcosmu, lsinmu, lcosnv, lsinnv, lcosmum, &
1158  lsinmum, lcosnvn, lsinnvn
1159 !-----------------------------------------------
1160 
1161  iout = 0 ! Assume first that target point not near edge/axis
1162  ss_in = ss*ohs + ihalf*0.5_dp + one
1163  ns_in = int(ss_in) ! Calculate surface closest target from below
1164  IF (ihalf == 1 .AND. ns_in == 1) ns_in = 2 ! Extrapolation needed on the half mesh near axis
1165  IF (ns_in == ns) ns_in = ns - 1 ! Extrapolation needed (almost always on half mesh) near edge
1166  ns_out = ns_in + 1 ! Surface closest to target from above
1167  d1 = (ss_in - ns_in) ! Coefficients for linear interpolation
1168  d2 = one - d1 ! xv(ss_in) = d1*xv(ns_out) + d2*xv(ns_in)
1169 
1170  IF (btest(dflag, b_ds)) THEN
1171  IF (ihalf == 0) THEN
1172  IF (ns_in == 1) iout = -1 ! Marks that point at axis for radial derivative
1173  ELSE
1174  IF (ns_in == 2) iout = -1 ! Marks that point at axis for radial derivative
1175  ENDIF
1176  IF (ns_in == ns - 1) iout = 1 ! Marks that point at edge for radial derivative
1177  ENDIF
1178 
1179  IF (btest(dflag, b_ds)) THEN
1180  ndum = 4
1181  ELSE
1182  ndum = 2
1183  ENDIF
1184 
1185  ALLOCATE(work0(0:mpol,-ntor:ntor,ndum), stat=istat)
1186  IF (istat .NE. 0) stop 'Alloc. problem in TOIJSP_1P'
1187  work0 = zero
1188 
1189  work0(:,:,1) = orthonorm(:,:)*xmn(:,:,ns_in)
1190  work0(:,:,2) = orthonorm(:,:)*xmn(:,:,ns_out)
1191  IF (btest(dflag, b_ds)) THEN
1192  IF (ihalf == 0) THEN
1193  IF (ns_in .GT. 1) work0(:,:,ndum-1) = &
1194  orthonorm(:,:)*xmn(:,:,ns_in-1) ! ...compute quantity at previous...
1195  ELSE
1196  IF (ns_in .GT. 2) work0(:,:,ndum-1) = &
1197  orthonorm(:,:)*xmn(:,:,ns_in-1)
1198  ENDIF
1199  IF (ns_out .LT. ns) work0(:,:,ndum) = &
1200  orthonorm(:,:)*xmn(:,:,ns_out+1) ! ...and next surfaces as well
1201  ENDIF
1202
