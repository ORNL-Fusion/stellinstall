265  FUNCTION restart_read_txt(filename, mpolin, ntorin, nsin)
266  USE quantities, ONLY: jbsupsmnsh, jbsupsmnch, &

297  presc(0:mpol,-ntor:ntor,ns), stat=istat)

299 
300  DO js=1,ns
301  DO m=0,mpol
302  DO n=-ntor,ntor
303  READ (15,300) bsupss(m,n,js), bsupuc(m,n,js), &
304  bsupvc(m,n,js), presc(m,n,js)
305  ENDDO
306  ENDDO
307  ENDDO
308 
309  IF (lasym) THEN
310  ALLOCATE (bsupsc(0:mpol,-ntor:ntor,ns), &
311  bsupus(0:mpol,-ntor:ntor,ns), &
312  bsupvs(0:mpol,-ntor:ntor,ns), &
313  press(0:mpol,-ntor:ntor,ns), stat=istat)

316  DO n=-ntor,ntor
317  READ (15,300) bsupsc(m,n,js), bsupus(m,n,js), &

319  ENDDO
320  ENDDO
321  ENDDO
322  END IF

336  CALL interpit(presc, jpmnch, ns, nsin, mpol, mpolin, ntor, ntorin)
337  jpmnch(m0,-nmin:nmin,1) = jpmnch(m0,-nmin:nmin,2)
338  DEALLOCATE (bsupss, bsupuc, bsupvc, presc, stat=istat)
339 
340  IF (lasym) THEN
341  jbsupsmnch(:,:,1) = 0
342  jbsupvmnsh(:,:,1) = 0
343  jpmnsh(:,:,1) = 0
344 
345  CALL interpit(bsupsc, jbsupsmnch, ns, nsin, mpol, mpolin, ntor, ntorin)
346  jbsupsmnch(m1,-nmin:nmin,1) = jbsupsmnch(m1,-nmin:nmin,2)
347  CALL interpit(bsupus, jbsupumnsh, ns, nsin, mpol, mpolin, ntor, ntorin)
348  jbsupumnsh(m1,-nmin:nmin,1) = jbsupumnsh(m1,-nmin:nmin,2)

793  SUBROUTINE interpit(aold, anew, ns_old, ns_new, &
794  & mpol_old, mpol_new, ntor_old, ntor_new)
795  USE stel_constants, ONLY: one, zero
796  IMPLICIT NONE
797 !-----------------------------------------------
798 ! D u m m y A r g u m e n t s
799 !-----------------------------------------------
800  INTEGER, INTENT(in) :: ns_old
801  INTEGER, INTENT(in) :: ns_new
802  INTEGER, INTENT(in) :: mpol_old
803  INTEGER, INTENT(in) :: mpol_new
804  INTEGER, INTENT(in) :: ntor_old
805  INTEGER, INTENT(in) :: ntor_new
806  REAL(dp), INTENT(in) :: aold(0:mpol_old,-ntor_old:ntor_old,ns_old)
807  REAL(dp), INTENT(out) :: anew(0:mpol_new,-ntor_new:ntor_new,ns_new)
808 !-----------------------------------------------
809 ! L o c a l V a r i a b l e s
810 !-----------------------------------------------
811  REAL(dp), PARAMETER :: p5 = 0.5_dp
812  REAL(dp), ALLOCATABLE :: temp1(:,:,:)
813  INTEGER :: mpol_min, ntor_min
814  INTEGER :: js_old, js_new, istat
815  REAL(dp) :: hs_old, hs_new, snew, sold, x1
816 !-----------------------------------------------
817 
818  mpol_min = min(mpol_old, mpol_new)
819  ntor_min = min(ntor_old, ntor_new)
820 
821  anew = 0
822 
823  IF (ns_old .EQ. ns_new) THEN
824  anew(0:mpol_min,-ntor_min:ntor_min,:) = aold(0:mpol_min,-ntor_min:ntor_min,:)
825  RETURN
826  END IF
827 
828  ALLOCATE (temp1(0:mpol_min,-ntor_min:ntor_min,ns_old), stat=istat)
829  CALL assert(istat.eq.0,'Allocation error #1 in INTERPIT')
830 !
831 ! FIRST INTERP TO FULL MESH FROM HALF-MESH
832 ! ASSUMES js=1 FULL-MESH VALUE VALUE IS STORED IN aold(1)
833 !
834  temp1(:,:,1) = 0
835  temp1(:,:,2:ns_old-1) = p5*(aold(:,:,2:ns_old-1) + aold(:,:,3:ns_old))
836  temp1(:,:,ns_old) = 2*aold(:,:,ns_old) - temp1(:,:,ns_old-1)
837  temp1(0:1,:,1) = 2*aold(0:1,:,2) - temp1(0:1,:,2)
838  hs_old = one/(ns_old-1)
839  hs_new = one/(ns_new-1)
840 
841  DO js_new=1, ns_new
842  snew = (js_new-1)*hs_new
843  js_old = 1 + int(snew/hs_old)
844  x1 = snew - (js_old-1)*hs_old
845  x1 = x1/hs_old
846  IF (x1 .GT. one) x1 = 1
847  IF (x1 .LT. zero) x1 = 0
848  CALL assert(js_old.le.ns_old,'js_old >= ns_old in INTERPIT')
849  IF (js_old .eq. ns_old) THEN
850  anew(:,:,js_new) = temp1(:,:,js_old)
851  ELSE
852  anew(:,:,js_new) = (one-x1)*temp1(:,:,js_old) &
853  + x1*temp1(:,:,js_old+1)
854  END IF
855  END DO
856 !
857 ! AVERAGE BACK TO HALF-MESH
858 !
859  DEALLOCATE (temp1)
860  ALLOCATE (temp1(0:mpol_min,-ntor_min:ntor_min,ns_new), stat=istat)
861  CALL assert(istat.eq.0,'Allocation error #2 in INTERPIT')
862  temp1 = anew
863  anew(:,:,1) = 0
864  anew(:,:,2:ns_new) = p5*(temp1(:,:,2:ns_new) + temp1(:,:,1:ns_new-1))
865  DEALLOCATE(temp1)
866 
867  END SUBROUTINE interpit
