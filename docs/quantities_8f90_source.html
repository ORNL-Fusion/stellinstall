17 !
18 ! Convention: The name of the variables may contain:
19 ! i. The first one or two letters tells the quantity. So, "p"=pressure, "B" = magnetic field,
20 ! "jB" = magnetic field*jacobian, "v" = velocity, "jv" = velocity*jacobian
21 ! ii. The next three letters maybe "SUB" or "SUP", to distinguish covariant/contravariant
22 ! ii. The next letter maybe "s", "u" or "v" to distinguish radial/poloidal/toroidal components
23 ! iii. The next two letters maybe "MN" or "IJ", to distinguish Fourier harmonics or angular values
24 ! iv. The next letter maybe "C" or "S" to distinguish COSINE or SINE parity
25 ! v. The last letter maybe "F" or "H" to distinguish FULL or HALF radial mesh
26 !
27 ! Radial meshes
28 ! velocity, force components :: full mesh
29 ! p, contravariant components of B :: half mesh
30 ! jacobh (jacobian to curvilinear coordinates) :: half mesh (stores <jacobh(2)> in jacobh(1))

89  SUBROUTINE init_quantities
90  USE fourier, ONLY: getorigin, tomnsp, toijsp, f_none, f_cos, &
91  f_sin, f_sum, f_full
92  USE metrics, ONLY: sqrtg
93  IMPLICIT NONE
94 !-----------------------------------------------
95 ! L o c a l V a r i a b l e s
96 !-----------------------------------------------
97  INTEGER, PARAMETER :: m0=0
98  INTEGER :: istat
99  REAL(dp) :: sum1
100 !-----------------------------------------------
101  w_factor = 1._dp/abs(wb_i)
102  p_factor = w_factor
103  b_factor = sqrt(w_factor)
104  gnorm_i = gnorm_i/w_factor
105 
106 !
107 ! Allocate and initialize dependent variables
108 !
109  CALL alloc_quantities
110 
111  jacobh(:,:,:) = reshape(sqrtg, shape(jacobh)) ! Reshapes sqrtg in local form
112 
113 !AVOID DIVIDE BY ZERO AND STORE AVERAGED VALUE (OVER THETA)
114 !AT FIRST 1/2 GRID PT AT ORIGIN IN JACOBH(1)
115  CALL getorigin (jacobh, m0, 0)
116  signjac = jacobh(1,1,2)/abs(jacobh(1,1,2))
117 
118 !TO PRESERVE UNPERTURBED (CONSTANT IN s) PART OF PRESSURE, NYQUIST-FILTER JACOBIAN
119 !ON: SPH 04-28-16
120 ! IF (.FALSE.) THEN !SET TO FALSE TO TURN THIS OFF
121  jacobf = jacobh
122  CALL tomnsp(jacobh, jbsupsmnsh, f_cos, f_full) !use jbsupsmnsh for dummy storage
123  IF (l_asym) THEN
124  CALL tomnsp(jacobh, jbsupsmnch, f_sin, f_full)
125  END IF
126  CALL toijsp(jbsupsmnsh, jacobh, f_none, f_cos, f_full)
127  jbsupsmnsh = 0
128  IF (l_asym) THEN
129  CALL toijsp(jbsupsmnch, jacobh, f_sum, f_sin, f_full)
130  jbsupsmnch = 0
131  END IF
132 
133  IF (any(jacobh*signjac .LE. 0)) stop 'FILTERED JACOBIAN CHANGED SIGN!'
134  sum1 = sum((jacobh(:,:,2:)-jacobf(:,:,2:))**2/ &
135  (jacobh(:,:,2:)+jacobf(:,:,2:))**2)
136  IF (iam .EQ. 0) print '(a,1pe8.2)', &
137  ' JACOBIAN SPECTRAL TRUNCATION ERROR: ', sqrt(sum1/SIZE(jacobh))
138 ! END IF
139 
140  CALL to_full_mesh(jacobh, jacobf, nuv)
141 !Must be consistent with variational form of pressure evolution equation (DO NOT USE 2pt EXTRAPOLATIONS!)
142  jacobf(:,:,1) = jacobh(:,:,1)
143  jacobf(:,:,ns)= jacobh(:,:,ns)
144 
145  ALLOCATE (vp_f(ns), stat=istat)
146  IF (istat .NE. 0) stop 'Allocation error in init_bcovar'
147  CALL surfaverage(vp_f, jacobf, 1, ns)
148 
149 !SANITY CHECK: COMPARE TO volume_i
150  sum1 = 0.5*hs_i*(sum(vp_f(2:ns)) + sum(vp_f(1:ns-1)))*signjac
151  sum1 = (twopi*twopi)*sum1
152  IF (abs((sum1-volume_i)/volume_i) .GT. 1.e-3_dp) THEN
153 
154  IF (iam .EQ. 0) print *,'VMEC VOLUME: ',volume_i,' THIS VOLUME: ', sum1
155 ! STOP 'Volume does not agree with VMEC input!'
156  END IF
157 
158  END SUBROUTINE init_quantities

161  SUBROUTINE init_fields
162  USE fourier, ONLY: orthonorm, tomnsp_par, f_sin, f_cos, f_half
163  USE metrics, ONLY: lmns_i, lmnc_i, sqrtg, iflipj
164  USE nscalingtools, ONLY: startglobrow, endglobrow
165  USE hessian, ONLY: gather_array
166  USE mpi_inc
167  IMPLICIT NONE
168 !-----------------------------------------------
169 ! L o c a l V a r i a b l e s
170 !-----------------------------------------------
171  INTEGER, PARAMETER :: m0=0, m1=1
172  INTEGER :: istat, m, n, nsmin, nsmax, nloc, js
173  REAL(dp) :: r0
174  REAL(dp), DIMENSION(:), ALLOCATABLE :: phiph, chiph
175  REAL(dp), DIMENSION(:,:), ALLOCATABLE :: pmnf
176  REAL(dp), DIMENSION(:,:,:), ALLOCATABLE:: presif, presih, &
177  tmps, tmpu, tmpv, tmpp
178 !-----------------------------------------------
179 
180 !
181 ! Initialize half-mesh covariant components of jac*Bsuph-i and jac*presh
182 !
183 
184 #if defined(SKS)
185  nsmin=max(1,startglobrow); nsmax=min(endglobrow,ns)
186 #else
187  nsmin = 1; nsmax = ns
188 #endif
189  jbsupsmnsh(:,:,nsmin:nsmax) = 0
190 
191  ALLOCATE(phiph(ns), chiph(ns))
192 
193  phipf_i = signjac*iflipj*phipf_i
194  chipf_i = signjac*iflipj*chipf_i
195 
196  phiph(2:ns) = (phipf_i(2:ns) + phipf_i(1:nsh))/2
197  chiph(2:ns) = (chipf_i(2:ns) + chipf_i(1:nsh))/2
198  phiph(1) = 0; chiph(1) = 0
199 
200 !SKIP THIS TO USE INTERPOLATED LAMBDA FROM VMEC
201 ! IF (l_asym) THEN
202 ! IF (iam .EQ. 0) PRINT *,'--- IMPLEMENT COMPUTE LMNC! ---'
203 ! ELSE
204  CALL recomputelambda(lmns_i, lmnc_i, jacobh, orthonorm, &
205  phiph, chiph, nsmin, nsmax)
206 ! ENDIF
207 
208  DO m = 0,mpol
209  DO n = -ntor, ntor
210  jbsupumnch(m,n,nsmin:nsmax) = phiph(nsmin:nsmax)*(-n*nfp*lmns_i(m,n,nsmin:nsmax))
211  jbsupvmnch(m,n,nsmin:nsmax) = phiph(nsmin:nsmax)*( m *lmns_i(m,n,nsmin:nsmax))
212  END DO
213  END DO
214 
215  jbsupumnch(m0,0,nsmin:nsmax) = chiph(nsmin:nsmax) / orthonorm(0,0)
216  jbsupvmnch(m0,0,nsmin:nsmax) = phiph(nsmin:nsmax) / orthonorm(0,0)
217 
218  r0 = hs_i/2
219  jbsupumnch(m1,:,1) = 0
220  jbsupvmnch(m1,:,1) = 0
221  jbsupsmnsh(m1,:,2) = r0*jbsupumnch(m1,:,2) ! Need to make jbsupsmnsh - r0*jbsupumnch = 0 at js=2
222 
223 !STORE m=0 component of jbsupvmnch at origin in half-grid js=1
224  jbsupvmnch(m0,:,1) = jbsupvmnch(m0,:,2)
225  jbsupvmnch(m1:,:,1) = 0
226 
227  jbsupumnch(:,:,nsmin:nsmax) = b_factor*jbsupumnch(:,:,nsmin:nsmax)
228  jbsupvmnch(:,:,nsmin:nsmax) = b_factor*jbsupvmnch(:,:,nsmin:nsmax)
229  IF (l_asym) THEN
230  jbsupsmnch = 0
231 
232  DO m = 0, mpol
233  DO n = -ntor, ntor
234  jbsupumnsh(m,n,:) = phiph(:)*( n*nfp*lmnc_i(m,n,1:ns))
235  jbsupvmnsh(m,n,:) = phiph(:)*(-m *lmnc_i(m,n,1:ns))
236  END DO
237  END DO
238 
239  jbsupumnsh(m1,:,1) = 0
240  jbsupvmnsh(m1,:,1) = 0
241  jbsupsmnch(m1,:,2) = -r0*jbsupumnsh(m1,:,2) ! Need to make jbsupsmnch + r0*jbsupumnsh = 0 at js=2
242 
243  jbsupumnsh = b_factor*jbsupumnsh
244  jbsupvmnsh = b_factor*jbsupvmnsh
245 
246 !STORE m=0 component of jbsupvmnch at origin in half-grid js=1
247  jbsupvmnsh(m0,:,1) = jbsupvmnsh(m0,:,2)
248  jbsupvmnsh(m1:,:,1) = 0
249  END IF
250 
251  DEALLOCATE (phiph, chiph, sqrtg, stat=istat)
252  IF (istat .NE. 0) stop "Deallocate error #1 in init_fields"
253 
254 !
255 ! Initialize half mesh pressure
256 !
257  ALLOCATE(presih(ntheta,nzeta,ns), &
258  presif(ntheta,nzeta,ns), stat=istat)
259  IF (istat .NE. 0) stop "Allocate error #1 in init_field"
260 
261  DO js = max(1,nsmin-1), nsmax
262  presif(:,:,js) = p_factor*presf_i(js)
263  END DO
264 
265  nloc = max(1,startglobrow-1)
266 
267  CALL to_half_mesh_par(presif(:,:,nloc:nsmax), &
268  presih(:,:,nloc:nsmax), nuv, nloc, nsmax)
269 
270 ! Evolve (internally) jacob*(real pressure)
271  presih(:,:,nsmin:nsmax) = presih(:,:,nsmin:nsmax) * jacobh(:,:,nsmin:nsmax)
272 
273  CALL tomnsp_par(presih, jpmnch, f_cos, f_half, nsmin, nsmax) ! Initialize pressure harmonics (half mesh)
274  jpmnch(:,:,1) = 0
275  jpmnch(m0,:,1) = jpmnch(m0,:,2)
276 
277  IF (l_asym) THEN
278  CALL tomnsp_par(presih, jpmnsh, f_sin, f_half, nsmin, nsmax)
279  jpmnsh(:,:,1) = 0
280  jpmnsh(m0,:,1) = jpmnsh(m0,:,2)
281  END IF
282 
283 
284  DEALLOCATE(presif, presih, stat=istat)
285  IF (istat .NE. 0) stop "Deallocate error #2 in init_fields"
286 
287 !
288 ! Gather jbsupmnX's, jpmn onto all processors (same as in update_state_par but for initial values)
289 !
290 #if defined(SKS)
291  ALLOCATE (tmps(0:mpol,-ntor:ntor,ns), tmpu(0:mpol,-ntor:ntor,ns),&
292  tmpv(0:mpol,-ntor:ntor,ns), tmpp(0:mpol,-ntor:ntor,ns),&
293  stat=istat)
294  IF (istat .NE. 0) stop 'Allocate error #2 in init_fields'
295 
296  CALL gather_array(jbsupsmnsh(:,:,nsmin:nsmax), tmps)
297  CALL gather_array(jbsupumnch(:,:,nsmin:nsmax), tmpu)
298  CALL gather_array(jbsupvmnch(:,:,nsmin:nsmax), tmpv)
299  CALL gather_array(jpmnch(:,:,nsmin:nsmax), tmpp)
300 
301  jbsupsmnsh = tmps
302  jbsupumnch = tmpu
303  jbsupvmnch = tmpv
304  jpmnch = tmpp
305 
306  IF (l_asym) THEN
307 
308  CALL gather_array(jbsupsmnch(:,:,nsmin:nsmax), tmps)
309  CALL gather_array(jbsupumnsh(:,:,nsmin:nsmax), tmpu)
310  CALL gather_array(jbsupvmnsh(:,:,nsmin:nsmax), tmpv)
311  CALL gather_array(jpmnsh(:,:,nsmin:nsmax), tmpp)
312 
313  jbsupsmnch = tmps
314  jbsupumnsh = tmpu
315  jbsupvmnsh = tmpv
316  jpmnsh = tmpp
317 
318  END IF
319 
320  DEALLOCATE (tmps, tmpu, tmpv, tmpp, stat=istat)
321  IF (istat .NE. 0) stop "Deallocate error #3 in init_fields"
322 #endif
323  END SUBROUTINE init_fields

326  SUBROUTINE recomputelambda(lmns, lmnc, jacobh, orthonorm, phiph, &
327  chiph, nsmin, nsmax)
328 !-----------------------------------------------
329  USE island_params, ns=>ns_i, nuv=>nuv_i, mnmax=>mnmax_i, &
330  nfp => nfp_i, ntheta => nu_i, nzeta => nv_i
331  USE metrics, ONLY: guu, guv, gvv
332 
333  IMPLICIT NONE
334  INTEGER, INTENT(IN) :: nsmin, nsmax
335  REAL(dp), INTENT(OUT) :: lmns(mnmax,ns), lmnc(mnmax,ns)
336  REAL(dp), INTENT(IN) :: phiph(ns), chiph(ns), orthonorm(mnmax), &
337  jacobh(nuv, ns)
338  INTEGER :: js, m, n, mp, np, mn, mnp, lk, lu, lv, &
339  info, isign, n2
340  REAL(dp), ALLOCATABLE :: amat(:,:), brhs(:,:), atemp(:), btemp(:)
341  REAL(dp), ALLOCATABLE :: cosmni(:,:), cosmnp(:,:), &
342  sinmni(:,:), sinmnp(:,:)
343  REAL(dp) :: ton, toff
344 !-----------------------------------------------
345  CALL second0(ton)
346 
347  lmns(:,1) = 0
348  IF (l_asym) lmnc(:,1) = 0
349 !
350 ! ALLOCATE cos(mu+nv) and sin(mu+nv)
351 !
352  n2 = 1
353  IF (l_asym) n2 = 2
354  ALLOCATE (atemp(nuv), btemp(nuv), cosmni(nuv,mnmax), &
355  cosmnp(nuv,mnmax), &
356  brhs(mnmax,n2), amat(mnmax*n2, mnmax*n2), stat=info)
357  IF (info .NE. 0) stop 'Allocation error in RecomputeLambda'
358  IF (l_asym) THEN
359  ALLOCATE (sinmni(nuv,mnmax), sinmnp(nuv,mnmax), &
360  stat=info)
361  IF (info .NE. 0) stop 'Allocation error in RecomputeLambda'
362  END IF
363 !COMPUTE MATRIX ELEMENTS, SURFACE BY SURFACE (can parallelize this: Sudip)
364 
365  mn = 0
366  DO n=-ntor,ntor
367  isign = 1; IF (n .LT. 0) isign = -1
368  np = abs(n)
369  DO m=0,mpol
370  mn = mn+1
371  lk = 0
372  DO lv = 1, nzeta
373  DO lu = 1, ntheta
374  lk = lk+1
375  cosmni(lk,mn) = cosmui(lu,m)*cosnv(lv,np) &
376  - isign*sinmui(lu,m)*sinnv(lv,np)
377  cosmnp(lk,mn) = cosmu(lu,m)*cosnv(lv,np) &
378  - isign*sinmu(lu,m)*sinnv(lv,np)
379  IF (.NOT. l_asym) cycle
380  sinmni(lk,mn) = sinmui(lu,m)*cosnv(lv,np) &
381  + isign*cosmui(lu,m)*sinnv(lv,np)
382  sinmnp(lk,mn) = sinmu(lu,m)*cosnv(lv,np) &
383  - isign*cosmu(lu,m)*sinnv(lv,np)
384  END DO
385  END DO
386  IF (m.EQ.0 .AND. n.LT.0) THEN
387  cosmni(:,mn) = 0
388  cosmnp(:,mn) = 0
389  IF (l_asym) THEN
390  sinmni(:,mn) = 0
391  sinmnp(:,mn) = 0
392  END IF
393  END IF
394  END DO
395  END DO
396 
397 !
398 ! COMPUTE Lambda from sqrt(g)*J^s == mB_v - nB_u = 0
399 !
400  DO js = max(nsmin,2), nsmax
401 
402  mn = 0
403  DO n=-ntor,ntor
404  DO m=0,mpol
405 !
406 ! RHS (driving) TERMS [sinmi, -cosmni parts]
407 !
408  mn = mn+1
409  btemp = (-m*(chiph(js)*guv(:,js)+phiph(js)*gvv(:,js)) &
410  + n*nfp*(chiph(js)*guu(:,js)+phiph(js)*guv(:,js))) &
411  / jacobh(:,js)
412  brhs(mn,1) = sum(cosmni(:,mn)*btemp)
413  IF (l_asym) THEN
414  brhs(mn,2) =-sum(sinmni(:,mn)*btemp)
415  END IF
416 
417 !
418 ! MATRIX ELEMENTS - COEFFICIENTS OF Lambda(mn,js)
419 !
420  mnp = 0
421 
422  DO np=-ntor,ntor
423  DO mp=0,mpol
424  mnp=mnp+1
425  atemp = (m*mp*gvv(:,js)+(n*nfp)*(np*nfp)*guu(:,js) &
426  - nfp*(m*np+mp*n)*guv(:,js))/jacobh(:,js)
427  amat(mn,mnp) = sum(cosmni(:,mn)*cosmnp(:,mnp)*atemp)
428  IF (mnp.EQ.mn .AND. amat(mn,mnp).EQ.zero) &
429  amat(mn,mnp) = signjac
430  IF (.NOT.l_asym) cycle
431  amat(mn,mnp+mnmax) =-sum(cosmni(:,mn)*sinmnp(:,mnp)*atemp)
432  amat(mn+mnmax,mnp) =-sum(sinmni(:,mn)*cosmnp(:,mnp)*atemp)
433  amat(mn+mnmax,mnp+mnmax) &
434  = sum(sinmni(:,mn)*sinmnp(:,mnp)*atemp)
435  IF (mnp.EQ.mn .AND. amat(mn+mnmax,mnp+mnmax).EQ.zero) &
436  amat(mn+mnmax,mnp+mnmax) = signjac
437  END DO
438  END DO
439  END DO
440  END DO
441 
442  CALL solver(amat, brhs, mnmax*n2, 1, info)
443  IF (info .NE. 0) stop 'INFO != 0 IN RECOMPUTELAMBDA'
444  lmns(:,js) = brhs(:,1)/orthonorm/phiph(js) !Divide by phiph -> old style
445  IF (l_asym) &
446  lmnc(:,js) = brhs(:,2)/orthonorm/phiph(js)
447 
448  END DO
449 
450  IF (mn.NE.mnmax .OR. mnp.NE.mnmax) stop 'mn or mnp != mnmax in RECOMPUTELAMBDA'
451 
452  DEALLOCATE(atemp, btemp, cosmni, cosmnp, brhs, amat)
453  IF (l_asym) DEALLOCATE(sinmni, sinmnp)
454 
455  CALL second0(toff)
456  IF (iam .EQ. 0) print 100,' TIME IN RECOMPUTELAMBDA: ', toff-ton,' S'
457 
458  100 FORMAT(a,f10.2,a)
459  END SUBROUTINE recomputelambda

463  SUBROUTINE alloc_quantities
464 !
465 ! Allocates variables that will be needed through the advancing subroutines.
466 ! More may be added in the future.
467 ! All variables allocated here are then deallocated in DEALLOC_QUANTITIES,
468 ! which should be called at the end of the run.
469 !
470  IMPLICIT NONE
471 !-----------------------------------------------
472 ! L o c a l V a r i a b l e s
473 !-----------------------------------------------
474  INTEGER :: istat
475 !-----------------------------------------------
476 ! Half mesh quantities (harmonics) evolved: 3 magnetic field components and pressure
