188  SUBROUTINE add_perturb (xc, getwmhd)
189  USE siesta_error
190  USE fourier, ONLY: tomnsp_par, f_cos, f_sin, f_none
191  IMPLICIT NONE
192 !-----------------------------------------------
193 ! D u m m y A r g u m e n t s
194 !-----------------------------------------------
195  REAL(dp) :: xc(:)
196  REAL(dp), EXTERNAL :: getwmhd
197 !-----------------------------------------------
198 ! L o c a l V a r i a b l e s
199 !-----------------------------------------------
200  INTEGER,PARAMETER :: ipert_sign=2
201  INTEGER :: js, l, iprint, mres0, nres0, isign1, icount, &
202  irscale, imin(2), jstart, istat, jsave
203  REAL(dp) :: w0, w1, wmhd, eta_factor_save, p_width_min
204  REAL(dp) :: normal, helpert0, hp, rad, chip0, phip0
205  LOGICAL :: lresist_save
206 !-----------------------------------------------
207 !
208 ! Add helical RESISTIVE flux perturbation on full mesh
209 ! Approximate nonideal E_sub ~ f(mu+nv) B_sub in update_bfield (E ~ B)
210 !
211  IF (ntor.EQ.0 .OR. all(helpert.EQ.zero)) RETURN
212 
213 ! INITIAL STATE: allocate buv_res so bsubXijf will be computed in init_state
214  nsmin=max(1,startglobrow-1); nsmax=min(ns,endglobrow+1)
215 ! nsmin = 1; nsmax = ns
216  ALLOCATE (buv_res(ntheta,nzeta,nsmin:nsmax), &
217  jdotb(ntheta,nzeta,nsmin:nsmax), stat=istat)
218  IF (istat .NE. 0) CALL errorabort('ALLOCATION ERROR IN add_perturbation')
219 
220  lresist_save = lresistive
221  lresistive=.false.
222  eta_factor_save = eta_factor
223  eta_factor=1
224  xc = 0
225  w0 = getwmhd(xc) !Get latest current and bsubXijf
226 
227  IF (iam .EQ. 0) THEN
228  DO iprint = 6, 33, 27
229  IF (.NOT.lverbose .AND. iprint.EQ.6) cycle
230  WRITE (iprint, '(/,a,/,2a)') &
231  ' Adding helical magnetic field perturbations', &
232  ' 10^6 X Del-W mres nres HelPert', &
233  ' rad |m*chip+n*phip| iota radial width'
234  END DO
235  ENDIF
236 
237 #if defined(JDOTB_PERT)
238 ! Compute jdotb to extract resonant components of sqrt(g)*J dot B
239  jdotb = ksupsijf0(:,:,nsmin:nsmax)*bsubsijf(:,:,nsmin:nsmax) &
240  + ksupuijf0(:,:,nsmin:nsmax)*bsubuijf(:,:,nsmin:nsmax) &
241  + ksupvijf0(:,:,nsmin:nsmax)*bsubvijf(:,:,nsmin:nsmax)
242 
243  ALLOCATE (bmnc_res(0:mpol,-ntor:ntor, nsmin:nsmax), &
244  bmns_res(0:mpol,-ntor:ntor, nsmin:nsmax), stat=istat)
245  IF (istat .NE. 0) CALL errorabort('ISTAT != 0 IN ADD_PERT')
246 
247  bmns_res = 0
248  CALL tomnsp_par(jdotb, bmnc_res, f_cos)
249  IF (lasym) CALL tomnsp_par(jdotb, bmns_res, f_sin)
250 #endif
251 
252 ! Compute responses to resonant resistitive perturbations
253  lresistive=.true.
254  normal = hs_i**2
255 
256  res_test: DO icount = 1, SIZE(helpert)
257  helpert0 = abs(helpert(icount)/normal)
258  mres0 = abs(mres(icount))
259  IF (helpert0.EQ.zero .OR. mres0.GT.mpol) cycle
260 
261 !Scan in radius to determine primary resonance
262 
263  nres_test: DO nres0=-ntor,ntor
264  jstart = 3 !Avoid 0/0 chip/phip at origin
265  jsurf_test: DO WHILE (jstart .LT. ns-1)
266 
267  IF (.NOT.findresonance(mres0, nres0, rad, jstart)) EXIT
268  w1 = 2*w0
269 
270  pert_sign: DO isign1 = 1, 2
271 
272 ! NOTE: in update_bfield, where perturbed b-field is computed, this is multiplied
273 ! by eta_prof = rho*(1-rho) so it vanishes at both ends. Multiply again
274 ! here by that factor to assure radial derivatives entering in dB also vanish
275 ! at endpoints s=0,1
276 
277  scale_test: DO irscale = 1, irad_scale
278  CALL getrespert(irscale, isign1, mres0, nres0, rad, &
279  helpert0, chip0, phip0)
280  xc = 0
281  wmhd = getwmhd(xc)
282  IF (wmhd .LT. w1) THEN
283  imin(1) = irscale
284  imin(2) = isign1
285  w1 = wmhd
286  p_width_min = p_width
287  ENDIF
288  END DO scale_test
289  END DO pert_sign
290 
291 ! Make sure energy decreases
292  wmhd = w1
293 ! recompute perturbation buv_res here
294  irscale = imin(1)
295  isign1 = imin(2)
296 
297  IF (iam .EQ. 0) THEN
298  hp = helpert0
299  IF (isign1 .EQ. 2) hp = -hp
300 
301  DO iprint = 6, 33, 27
302  IF (.NOT.lverbose .AND. iprint.EQ.6) cycle
303  WRITE (iprint, 100) 10**6*(wmhd-w0), mres0, nres0, &
304  hp*normal, rad, abs(mres0*chip0+nres0*nfp*phip0), &
305  chip0/phip0, p_width_min
306  END DO
307  100 FORMAT(1p,e12.3,2(i8),3x,1pe10.2,0p,f8.2,f11.2,4x,2(f11.2))
308  CALL flush(33)
309  END IF
310 
311  CALL getrespert(irscale, isign1, mres0, nres0, rad, &
312  helpert0, chip0, phip0)
313 
314  wmhd = getwmhd(xc)
315  IF (abs(wmhd-w1) .GT. 1.e-12_dp) THEN
316  CALL siesta_error_set_error(siesta_error_general, 'Error1 in Perturb')
317  END IF
318  xc = 0
319  CALL update_state(.false., zero, zero)
320 !Compute dW relative to perturbed state
321  w0 = wmhd
322 
323  END DO jsurf_test ! Look for multiple resonances
324  END DO nres_test
325  END DO res_test
326 
327  IF (iam .EQ. 0) THEN
328  DO iprint = 6, 33, 27
329  IF (.NOT.lverbose .AND. iprint.EQ.6) cycle
330  WRITE (iprint, *)
331  END DO
332  END IF
333 
334  DEALLOCATE (buv_res, jdotb, bmnc_res, bmns_res, stat=istat)
335  lresistive = lresist_save
336  eta_factor = eta_factor_save
337 
338  END SUBROUTINE add_perturb

341  LOGICAL FUNCTION findresonance(mres0, nres0, resrad, jstart)
342  IMPLICIT NONE
343 !-----------------------------------------------
344 ! D u m m y A r g u m e n t s
345 !-----------------------------------------------
346  INTEGER, INTENT(in) :: mres0
347  INTEGER, INTENT(in) :: nres0
348  INTEGER, INTENT(inout) :: jstart
349  REAL(dp), INTENT(out) :: resrad
350 !-----------------------------------------------
351 ! L o c a l V a r i a b l e s
352 !-----------------------------------------------
353  INTEGER :: js, jmin
354  REAL(dp) :: del1, del2, rho1, rho2, delmin
355 !-----------------------------------------------
356 !
357 ! Starting at jstart, find place where mres0*chip+n*nfp*phip is minimum (changes sign)
358 !
359  del2 = 0
360  resrad = 0
361  delmin = -1
362  IF (mres0*nres0 .EQ. 0) THEN
363  findresonance=.false.; RETURN
364  END IF
365 
366 ! Check for zero crossing
367  DO js = jstart, ns-1
368  del1 = mres0*chipf_i(js)+nres0*nfp*phipf_i(js)
369  del1 = del1/max(abs(mres0*chipf_i(js)),abs(nres0*nfp*phipf_i(js)),1.e-20_dp)
370  IF (delmin.EQ.-1 .OR. abs(del1).LT.delmin) THEN
371  IF (delmin .EQ. -1) del2 = del1
372  delmin = abs(del1)
373  jmin = js
374  END IF
375 
376  IF (del1*del2 < zero) THEN
377  jstart = js+1
378  rho2 = hs_i*(js-2)
379  rho1 = hs_i*(js-1)
380  resrad = (rho2*abs(del1)+rho1*abs(del2))/(abs(del1)+abs(del2))
381  delmin=0
382  EXIT
383  END IF
384  del2 = del1
385  END DO
386 
387 !IF NO ZERO-CROSSING, RESONANCE MIGHT BE VERY CLOSE
388  IF (delmin .LT. 1.e-3_dp) THEN
389  jstart = jmin+1
390  resrad = hs_i*(jmin-1)
391  findresonance=.true.
392  ELSE
393  jstart=ns-1
394  resrad = 0
395  findresonance=.false.
396  END IF
397 
398  END FUNCTION findresonance

401  SUBROUTINE getrespert(irscale, isign1, mres0, nres0, rad, &
402  helpert0, chip0, phip0)
403  USE fourier, ONLY: toijsp_par, toijsp2, f_cos, f_sin, f_sum, f_none
404  IMPLICIT NONE
405 !-----------------------------------------------
406 ! D u m m y A r g u m e n t s
407 !-----------------------------------------------
408  INTEGER, INTENT(in) :: irscale, isign1
409  INTEGER, INTENT(in) :: mres0, nres0
410  REAL(dp), INTENT(in) :: rad
411  REAL(dp), INTENT(out) :: chip0, phip0
412  REAL(dp), INTENT(in) :: helpert0
413 !-----------------------------------------------
414 ! L o c a l V a r i a b l e s
415 !-----------------------------------------------
416  INTEGER :: js, istat
417  REAL(dp) :: rho, rhores, pert_prof(ns), help_local, locrad
418  REAL(dp), ALLOCATABLE :: bmn_res(:,:)
419  REAL(dp), ALLOCATABLE :: resc(:,:,:), ress(:,:,:)
420  REAL(dp), ALLOCATABLE :: temp_uv(:,:,:)
421 !-----------------------------------------------
422  IF (mres0.GT.mpolin .OR. abs(nres0).GT.ntorin) RETURN
423 
424  help_local = helpert0
425  IF (isign1 .EQ. 2) help_local = -helpert0
426 
427 ! Compute radial form factor (window-function)
428  js = int(rad/hs_i) + 1
429  IF (js.LT.1 .OR. js.GT.ns) RETURN
430  chip0 = chipf_i(js)
431  phip0 = phipf_i(js)
432  locrad = REAL(irscale-1,dp)/(irad_scale-1)
433  p_width = 0.1_dp + 0.9_dp*locrad !decay length, ~delta-fcn for irscale=1 to broad (full radius)
434  locrad = one/p_width
435  DO js = 1,ns
436  rho = hs_i*(js-1)
437  rhores = rho-rad
438  rhores = locrad*rhores
439  pert_prof(js) = one/(one+rhores*rhores) !tearing parity; mult by alpha for equal area
440 ! pert_prof(js) = rhores*pert_prof(js) !odd-parity (rippling mode)
441  END DO
442 
443 !THIS DOESN'T SEEM TO WORK AS WELL AS THE SIMPLE FORM FACTOR
444 !SO KEEP JDOTB_PERT UNDEFINED UNTIL WE CAN IMPROVE IT
445 #if defined(JDOTB_PERT)
446  ALLOCATE(resc(0:mpol,-ntor:ntor, nsmin:nsmax), &
447  ress(0:mpol,-ntor:ntor, nsmin:nsmax), stat=istat)
448 
449  rho = max(maxval(abs(bmnc_res(mres0,nres0,:))), &
450  maxval(abs(bmns_res(mres0,nres0,:))))
451 
452  resc = 0; ress = 0
453  resc(mres0,nres0,:) = help_local*bmnc_res(mres0,nres0,:)
454 
455  CALL toijsp_par(resc, buv_res, f_none, f_cos)
456 
457  IF (lasym) THEN
458 !NOTE TO MRC: eventually use separate HelPertS here
459  ress(mres0,nres0,:) = help_local*bmns_res(mres0,nres0,:)
460  CALL toijsp_par(ress, buv_res, f_sum, f_sin)
461  END IF
462 
463  IF (rho .NE. zero) buv_res = buv_res/rho
464 
465 ! Apply radial form-factor
466  DO js = nsmin, nsmax
467  buv_res(:,:,js) = buv_res(:,:,js)*pert_prof(js)
468  END DO
469 
470  DEALLOCATE (resc, ress, stat=istat)
471 #else
472  ALLOCATE (bmn_res(0:mpol,-ntor:ntor), stat=istat)
473  IF (istat .NE. 0) CALL errorabort('ISTAT != 0 IN GETRESPERT')
474  bmn_res = 0
475 
476  bmn_res(mres0,nres0) = help_local
477  CALL toijsp2(bmn_res, buv_res, f_cos, nsmin)
478 
479  IF (lasym) THEN
480  ALLOCATE (temp_uv(ntheta,nzeta,1), stat=istat)
481 !NOTE TO MRC: eventually use separate HelPertS here
482  bmn_res(mres0,nres0) = help_local
483  CALL toijsp2(bmn_res, temp_uv, f_sin, 1)
484  buv_res(:,:,nsmin) = buv_res(:,:,nsmin) + temp_uv(:,:,1)
485  DEALLOCATE(temp_uv)
486  END IF
487  DO js = nsmin+1, nsmax
488  buv_res(:,:,js) = buv_res(:,:,nsmin)*pert_prof(js)
489  END DO
490 
491  buv_res(:,:,nsmin) = buv_res(:,:,nsmin)*pert_prof(nsmin)
492 
493  DEALLOCATE (bmn_res, stat=istat)
494 #endif
495 
496  END SUBROUTINE getrespert

523 !-----------------------------------------------
