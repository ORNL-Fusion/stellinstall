340  FUNCTION pprofile_get_value(this, s_arg)
341  USE line_segment
342  USE functions
343 
344  IMPLICIT NONE
345 
346 ! Declare Arguments
347  REAL (rprec) :: pprofile_get_value
348  type(pprofile_class), INTENT (in) :: this
349  REAL (rprec), INTENT(in) :: s_arg
350 
351 ! local variables
352  REAL (rprec) :: s_use
353  REAL (rprec) :: s_01
354  LOGICAL :: l_01
355  INTEGER :: i
356  INTEGER :: iflag
357  REAL (rprec) :: start_time
358 
359 ! Start of executable code
360  start_time = profiler_get_start_time()
361 
362 ! Define argument and logical variable for s between zero and one
363  s_01 = max(zero, min(one,s_arg))
364  l_01 = (s_arg .ge. zero) .and. (s_arg .le. one)
365 
366  SELECT CASE(this%p_type)
367 
368  CASE (pprofile_none_type)
369  pprofile_get_value = -1.e49_rprec
370 
371 ! v3fit uses an extra varible as an offset than vmec as a result the b array is
372 ! one index shorter than what two_power takes.
373  CASE (pprofile_two_power_type)
374  pprofile_get_value = this%b(0)
375  IF (l_01) THEN
376  pprofile_get_value = pprofile_get_value &
377  & + two_power(s_arg, this%b(1:iub_b))
378  ENDIF
379 
380 ! v3fit uses an extra varible as an offset than vmec as a result the b array is
381 ! one index shorter than what two_power takes. This profile reverses the s
382 ! argument so that the maximum amplitude is at s=1 and the minimum is at s = 0.
383  CASE (pprofile_two_power_r_type)
384  pprofile_get_value = this%b(0)
385  IF (l_01) THEN
386  s_use = 1.0 - s_arg
387  pprofile_get_value = pprofile_get_value &
388  & + two_power(s_use, this%b(1:iub_b))
389  ENDIF
390 
391 ! v3fit uses an extra varible as an offset than vmec as a result the b array is
392 ! one index shorter than what two_power_gs takes. As a result only 5 guassian
393 ! peaks maybe specified instead of the 6 vmec can use.
394  CASE (pprofile_two_power_gs_type)
395  pprofile_get_value = this%b(0)
396  IF (l_01) THEN
397  pprofile_get_value = pprofile_get_value &
398  & + two_power_gs(s_arg, &
399  & this%b(1:iub_b))
400  ENDIF
401 
402  CASE (pprofile_power_series_type)
403  pprofile_get_value = zero
404  DO i = iub_b, ilb_b, -1 ! all of b array, backwards
405  pprofile_get_value = pprofile_get_value*s_01 &
406  & + this%b(i)
407  END DO
408 
409  CASE (pprofile_cubic_spline_type)
410  s_use = min(this%as(this%maxSplineIndex), &
411  & max(s_arg, this%as(1)))
412  CALL spline_cubic(s_use, pprofile_get_value, &
413  & this%as, this%af, &
414  & this%maxSplineIndex, iflag)
415  IF (iflag .ge. 0) THEN
416 
417  CALL profiler_set_stop_time('pprofile_get_value', &
418  & start_time)
419 
420  RETURN
421  ELSE IF (iflag .eq. -1) THEN
422  WRITE (*,*) 'ERROR: pprofile: outside value from ' // &
423  & 'spline_cubic'
424  ELSE IF(iflag .eq. -2) THEN
425  WRITE (*,*) 'ERROR: pprofile: decreasing s values ' // &
426  & 'in spline_cubic'
427  ELSE
428  WRITE (*,*) 'ERROR: pprofile: unknown error from ' // &
429  & 'spline_cubic'
430  ENDIF
431 ! Splines shuld not reach this point without triggering an error.
432  CALL exit(1)
433 
434  CASE (pprofile_akima_spline_type)
435  s_use = min(this%as(this%maxSplineIndex), &
436  & max(s_arg,this%as(1)))
437  CALL spline_akima(s_use, pprofile_get_value, &
438  & this%as, this%af, &
439  & this%maxSplineIndex, iflag)
440  IF (iflag < 0) THEN
441  WRITE (*,*) 'ERROR: pprofile: bad value from ' // &
442  & 'spline_akima requested'
443  CALL exit(1)
444  ENDIF
445 
446  CASE (pprofile_line_segment_type, &
447  & pprofile_gp_1d_sexp_type, &
448  & pprofile_gp_1d_ln_sexp_type)
449  CALL line_seg(s_arg, pprofile_get_value, this%as, this%af, &
450  & this%maxSplineIndex)
451 
452  END SELECT
453 
454  CALL profiler_set_stop_time('pprofile_get_value', start_time)
455 
456  END FUNCTION
