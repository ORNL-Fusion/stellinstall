1094  SUBROUTINE v3fit_input_read_namelist(namelist_file)
1095  USE safe_open_mod
1096  USE v3_utilities, only: err_fatal
1097 
1098  IMPLICIT NONE
1099 
1100 ! Declare Arguments
1101  CHARACTER (len=*), INTENT(in) :: namelist_file
1102 
1103 ! local variables
1104  INTEGER :: iou_mnli, status
1105  INTEGER :: j, i
1106  INTEGER :: imin, imax
1107  REAL (rprec) :: start_time
1108 
1109 ! Start of executable code
1110  start_time = profiler_get_start_time()
1111 
1112 ! Initalize a default value of the I\O unit. V3FIT increments from there.
1113  iou_mnli = 0
1114  CALL safe_open(iou_mnli, status, trim(namelist_file), &
1115  & 'old', 'formatted')
1116  CALL assert_eq(0, status, 'v3fit_input_read_namelist' // &
1117  & ': Safe_open of ' // trim(namelist_file) // ' failed')
1118 
1119 ! Read the namelist input file.
1120  READ (iou_mnli, nml=v3fit_main_nli)
1121  CLOSE (iou_mnli, iostat=status)
1122  CALL assert_eq(0, status, 'v3fit_input_read_namelist' // &
1123  & ': Error closing ' // trim(namelist_file) // ' failed')
1124 
1125 ! Toggle use coil signals for the magnetics from the specification
1126  DO j = 1, SIZE(mag_spec_imin)
1127  IF (mag_spec_imin(j) .le. 0) EXIT
1128  imin = max(min(mag_spec_imin(j), v3fit_max_diagnostics), 1)
1129  imax = min(max(mag_spec_imax(j), imin), v3fit_max_diagnostics)
1130  mag_a(imin:imax) = mag_spec_use_induced(j)
1131  END DO
1132 
1133 ! Set weights from weight specification.
1134  DO j = 1, SIZE(sdo_w_spec_imin)
1135  IF (sdo_w_spec_imin(j) .le. 0) EXIT
1136  imin = max(min(sdo_w_spec_imin(j), v3fit_max_signals), 1)
1137  imax = min(max(sdo_w_spec_imax(j), imin), v3fit_max_signals)
1138  sdo_weight_a(imin:imax) = sdo_w_spec_weight(j)
1139  END DO
1140 
1141 ! Set sigmas from the sigma specification.
1142  DO j = 1,SIZE(sdo_s_spec_imin)
1143  IF (sdo_s_spec_imin(j) .le. 0) EXIT
1144  imin = max(min(sdo_s_spec_imin(j), v3fit_max_signals), 1)
1145  imax = min(max(sdo_s_spec_imax(j), imin), v3fit_max_signals)
1146  DO i = imin, imax
1147  IF ((sdo_sigma_a(i) .eq. 0) .or. &
1148  & sdo_spec_can_overwrite) THEN
1149  IF (sdo_s_spec_floor(j) .ge. 0) THEN
1150  sdo_sigma_a(i) = max(sdo_s_spec_floor(j), &
1151  & abs(sdo_s_spec_fraction(j)*sdo_data_a(i)))
1152  ELSE
1153  sdo_sigma_a(i) = sqrt(sdo_s_spec_floor(j)**2.0 + &
1154  & (sdo_s_spec_fraction(j)*sdo_data_a(i))**2.0)
1155  END IF
1156  END IF
1157  END DO
1158  END DO
1159 
1160  IF (main_nli_filename .eq. '') THEN
1161  main_nli_filename = namelist_file
1162  END IF
1163 
1164  IF ((r_major_radius .gt. 0.0) .and. &
1165  & (a_minor_radius .gt. 0.0)) THEN
1166  IF (n_lif + 1 .le. v3fit_max_geometrics) THEN
1167  n_lif = n_lif + 1 ! Add this as a new limiter.
1168  lif_arz(n_lif,:,:) = 0.0
1169  lif_arz(n_lif,0,0) = -a_minor_radius**2.0
1170  lif_arz(n_lif,2,0) = 1.0
1171  lif_arz(n_lif,0,2) = 1.0
1172  lif_rc(n_lif) = r_major_radius
1173  lif_zc(n_lif) = 0.0
1174  lif_sigma(n_lif) = 0.001_rprec
1175  n_phi_lif(n_lif) = v3fit_max_lif_size
1176  DO i = 1, v3fit_max_lif_size
1177  lif_phi_degree(n_lif,i) = (i - 1)*360.0 &
1178  & / v3fit_max_lif_size
1179  END DO
1180  lif_on_edge(n_lif) = .true.
1181  ELSE
