745  FUNCTION signal_get_e(this, a_model, use_cache, last_value)
746  USE model
747 
748  IMPLICIT NONE
749 
750 ! Declare Arguments
751  REAL (rprec) :: signal_get_e
752  type(signal_class), INTENT(inout) :: this
753  type(model_class), INTENT(in) :: a_model
754  LOGICAL, INTENT(in) :: use_cache
755  REAL (rprec), DIMENSION(4), INTENT(in) :: last_value
756 
757 ! local Variables
758  REAL (rprec), DIMENSION(4) :: modeled_signal
759  REAL (rprec), DIMENSION(4) :: modeled_sigma
760  REAL (rprec) :: observed_signal
761  REAL (rprec) :: start_time
762 
763 ! Start of executable code
764  start_time = profiler_get_start_time()
765 
766  observed_signal = signal_get_observed_signal(this, a_model)
767 
768  modeled_signal = signal_get_modeled_signal(this, a_model, &
769  & modeled_sigma, &
770  & use_cache, last_value)
771 
772 ! e = SQRT(W)*(O - M)/SQRT(sigma_o^2 + sigma_m^2)
773 ! Users are can be lax when specifiying the sigma for signals weighted to zero.
774 ! Avoid a possible divide by zero by error checking the weight first. If the
775 ! weight is zero, set the error vector explicitly to zero.
776  IF (this%weight .eq. 0.0) THEN
777  signal_get_e = 0.0
778  ELSE
779  signal_get_e = (observed_signal - modeled_signal(1)) &
780  & * sqrt(this%weight/signal_get_sigma2(this))
781  END IF
782 
783  CALL profiler_set_stop_time('signal_get_e', start_time)
784 
785  END FUNCTION
