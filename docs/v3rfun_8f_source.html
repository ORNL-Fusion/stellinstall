797 
798  ALLOCATE(xcart_s_grid_e(n_u, kp_shell, n_field_periods, 3))

812 !$OMP& SCHEDULE(STATIC)

814  phi0 = (k - 1)*delp

817  phi_grid_e(k,l) = xcyl(2)
818  DO i = 1,ir
819  xcyl(1) = rmin + (i - 1)*delr
820  DO j = 1,jz
821  xcyl(3) = zmin + (j - 1)*delz
822  xcart_grid_e(i,j,k,l,1:3) = cyl_to_cart(xcyl)
823  END DO ! over j
824  END DO ! over i
825  END DO ! over l
826  END DO ! over k
827 !$OMP END DO
828 !$OMP END PARALLEL
829

831 ! computed in the previous loop.
832  IF (use_con_shell) THEN

834 !$OMP& DEFAULT(SHARED)
835 !$OMP& PRIVATE(j,k,l,xcyl,u,phi0)
836 !$OMP DO
837 !$OMP& SCHEDULE(STATIC)
838  DO k = 1,kp_shell
839  phi0 = (k - 1)*delp_shell
840  DO l = 1,n_field_periods
841  xcyl(2) = phi0 + (l - 1)*fperiod
842  phi_grid_shell(k,l) = xcyl(2)
843  DO j = 1,n_u
844  u = (j - 1)*delu
845  xcyl(1) = major_radius + minor_radius*cos(u)
846  xcyl(3) = minor_radius*sin(u)
847  xcart_s_grid_e(j,k,l,1:3) = cyl_to_cart(xcyl)
848  END DO
849  END DO
850  END DO
851 !$OMP END DO
852 !$OMP END PARALLEL
853  END IF

867  & 'formatted')

924 ! Parallelize the outer most loops. Any local variables inside the loop must
925 ! be private.
926 !$OMP PARALLEL
927 !$OMP& DEFAULT(SHARED)
928 !$OMP& PRIVATE(i,j,k,l,acart)
929 !$OMP DO
930 !$OMP& SCHEDULE(STATIC)
931  DO i = 1,ir
932  DO j = 1,jz

934  pl_response(i,j,k,1:3) = zero
935  DO l = 1, n_field_periods

937  & xcart_grid_e(i,j,k,l,1:3), acart)

939  & cart2cyl_v(acart, phi_grid_e(k,l))
940  END DO ! l, field periods
941  pl_response(i,j,k,1:3) = pl_response(i,j,k,1:3) * &

943  END DO ! k

947 !$OMP END PARALLEL

949 ! When using stellator symmetry, only half of the field period needs to be

951 ! the symmetric point below the midplane. The counter point to point j is found
952 ! from by counting the grid in reverse.
953  IF (lstell_sym) THEN
954  DO i = 1,ir
955  iss = i
956  DO j = 1,jz

958  DO k = 1,kp_store

961  ELSE
962  kss = kp + 2 - k
963  ENDIF

965  & pl_response(iss,jss,kss,1)

967  & pl_response(iss,jss,kss,2)
968  pl_response_ss(i,j,k,3) = pl_response(i,j,k,3) + &
969  & pl_response(iss,jss,kss,3)
970  END DO ! k
971  END DO ! j
972  END DO ! i
973  END IF ! lstell_sym

975 ! Conducting Shell Response ---------------------------------------------------
976  IF (use_con_shell) THEN
977 
978 ! Parallelize the outer most loops. Any local variables inside the loop must
979 ! be private.
980 !$OMP PARALLEL
981 !$OMP& DEFAULT(SHARED)
982 !$OMP& PRIVATE(j,k,l,acart)
983 !$OMP DO
984 !$OMP& SCHEDULE(STATIC)

990  & xcart_s_grid_e(j,k,l,1:3), acart)
991  s_response(j,k,1:3) = s_response(j,k,1:3) + &
992  & cart2cyl_v(acart, phi_grid_shell(k,l))
993  END DO
994  END DO
995  END DO
996 !$OMP END DO
997 !$OMP END PARALLEL
998  s_response = s_response*mddc_desc_array(idc)%flux_factor
999 
1000 ! When using stellator symmetry, only half of the field period needs to be
1001 ! stored due to the symmetry. A poloidal point will have the same current as a
1002 ! the symmetric point below the midplane. The j=1 point corresponds to the
1003 ! same u position on the opposite side of the field perior. For j!=0, the
1004 ! symmetry point is found by counting indices backwards.
1005  IF (lstell_sym) THEN
1006  DO j = 1, n_u
1007  IF (j .eq. 1) THEN
1008  jss = 1
1009  ELSE
1010  jss = n_u + 2 - j
1011  END IF
1012  DO k = 1, kp_shell_store
1013  IF (k .eq. 1) THEN
1014  kss = 1
1015  ELSE
1016  kss = kp_shell + 2 - k
1017  END IF
1018 
1019  s_response_ss(j,k,1) = s_response(j,k,1) &
1020  & - s_response(jss,kss,1)
1021  s_response_ss(j,k,2) = s_response(j,k,2) &
1022  & + s_response(jss,kss,2)
1023  s_response_ss(j,k,3) = s_response(j,k,3) &
1024  & + s_response(jss,kss,3)
1025  END DO
1026  END DO
1027  END IF
1028  END IF

1031  IF (lstell_sym) THEN

1033  a_f => pl_response_ss(:,:,:,2)
1034  a_z => pl_response_ss(:,:,:,3)
1035  IF (use_con_shell) THEN
1036  a_s_r => s_response_ss(:,:,1)
1037  a_s_f => s_response_ss(:,:,2)
1038  a_s_z => s_response_ss(:,:,3)
1039  END IF
1040  ELSE
1041  a_r => pl_response(:,:,:,1)
1042  a_f => pl_response(:,:,:,2)
1043  a_z => pl_response(:,:,:,3)
1044  IF (use_con_shell) THEN
1045  a_s_r => s_response(:,:,1)
1046  a_s_f => s_response(:,:,2)
1047  a_s_z => s_response(:,:,3)
1048  END IF
1049  ENDIF

1052  response => magnetic_response_construct(code_name, date_run, &
1053  & trim(name_coils_dot), &
1054  & rdiag_coilg_1, extcur, &
1055  & kp, kp_shell, &
1056  & rmin, rmax, zmin, zmax, &
1057  & n_field_periods, &
1058  & lstell_sym, &
1059  & a_r, a_f, a_z, &
1060  & a_s_r, a_s_f, a_s_z, &
1061  & 0.0_dp)

1067  & trim(adjustl(mddc_desc_array(idc)%mdcoil%s_name)) // &
1068  & '_mdsig.nc'
1069  ELSE
1070  temp_filename = &
1071  & trim(adjustl(mddc_desc_array(idc)%mdcoil%s_name)) // &
1072  & '_' // trim(myid) // '_mdsig.nc'
1073  ENDIF
1074  status = 0
1075  CALL cdf_open(iou_mdsig_ncdf, trim(temp_filename), 'w', status)
1076  CALL assert_eq(0, status, &
1077  & 'write_mdsigs mdsig file open status')

1080  CALL cdf_define(iou_mdsig_ncdf, 'diagnostic_desc_d_type', &

1161 ! Subroutine to convert a tangent vector from Cartesian coordinates
1162 ! to cylindrical coordinates
1163  FUNCTION cart2cyl_v(vcart, xphi)
1164  USE stel_kinds
1165 
1166  IMPLICIT NONE
1167 
1168 ! Declare Arguments
1169  REAL (rprec), DIMENSION(3) :: cart2cyl_v
1170  REAL (rprec), DIMENSION(3), INTENT(in) :: vcart
1171  REAL (rprec), INTENT(in) :: xphi
1172 
1173 ! local variables
1174  REAL (rprec) :: cphi
1175  REAL (rprec) :: sphi
1176 
1177 ! Start of executable code
1178  cphi = cos(xphi)
1179  sphi = sin(xphi)
1180  cart2cyl_v(1) = vcart(1)*cphi + vcart(2)*sphi
1181  cart2cyl_v(2) = -vcart(1)*sphi + vcart(2)*cphi
1182  cart2cyl_v(3) = vcart(3)
1183 
1184  END FUNCTION
