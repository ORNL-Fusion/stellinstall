156 ntor=ntor_in
157 m=ndims*(mpol+1)*(2*ntor+1)

171

173  rightproc=rank+1

175  rightproc=mpi_proc_null

193  nbuffsize=nbuffsize*m

197  CALL mpi_barrier(siesta_comm,mpi_err)

202  WRITE(7000+rank,*) '>>> MPI_BSEND_OVERHEAD : ', mpi_bsend_overhead; CALL flush(7000+rank)
203  WRITE(7000+rank,*) '>>> nBuffsize : ', nbuffsize; CALL flush(7000+rank)
204  END IF

236 !------------------------------------------------
237 ! This computes the mapping of the block rows to
238 ! the processors. Computes the processors domain
239 ! boundaries in terms of global block indices.
240 !------------------------------------------------
241 SUBROUTINE bcyclicmapping
242 
243 !INTEGER, INTENT(OUT) :: startblock, endblock
244 INTEGER :: startblock, endblock
245 INTEGER :: lload, sload, myload
246 INTEGER :: numl, nums
247 INTEGER :: r, c
248 LOGICAL :: lnum, lnumg=.true.
249 
250 CALL assert(initremapdone, &
251 'Calling bcyclicMapping routine without calling initRemap')
252 
253 IF (mappingdone) RETURN
254 
255 lload=ceiling(REAL(n)/activeranks)
256 sload=floor(REAL(n)/activeranks)
257 
258 IF (lload.EQ.sload) THEN
259  myload=lload
260 ELSE
261  IF (rank.LT.mod(n,activeranks)) THEN
262  myload=lload
263  ELSE
264  myload=sload
265  END IF
266 END IF
267 
268 IF (sload.EQ.lload) THEN
269  nums=0
270  numl=rank
271 ELSE
272  IF (myload.EQ.lload) THEN
273  numl=rank
274  nums=0
275  ELSE
276  numl=mod(n,activeranks)
277  nums=rank-numl
278  END IF
279 END IF
280 
281 IF (rank.LT.activeranks) THEN !active ranks
282  startblock=numl*lload+nums*sload
283  endblock=startblock+myload-1
284 ELSE !idle ranks
285  startblock=-2
286  endblock=-3
287 END IF
288 
289 ! Fortranized indices
290 startblock=startblock+1
291 endblock=endblock+1
292 
293 CALL mpi_allgather(startblock,1,mpi_integer,bcyclicstartblockproc,&
294 &1,mpi_integer,siesta_comm,mpi_err)
295 
296 CALL mpi_allgather(endblock,1,mpi_integer,bcyclicendblockproc,&
297 &1,mpi_integer,siesta_comm,mpi_err)
298 
299 startglobrow=bcyclicstartblockproc(rank+1)
300 endglobrow=bcyclicendblockproc(rank+1)
301 
302 firstpartition_ge_2=.false.
303 IF((bcyclicendblockproc(1)-bcyclicstartblockproc(1)+1).GE.2) firstpartition_ge_2=.true.
304 
305 lastpartition_ge_2=.false.
306 IF((bcyclicendblockproc(activeranks)-bcyclicstartblockproc(activeranks)+1).GE.2) lastpartition_ge_2=.true.
307 
308 numblocks=bcyclicendblockproc(rank+1)-bcyclicstartblockproc(rank+1)+1
309 
310 !This may only occur on SOME processors, not all, so send to all
311 lnum = numblocks.GE.2
312 CALL mpi_reduce(lnum, lnumg, 1, mpi_logical, mpi_land, 0, siesta_comm, &
313  mpi_err)
314 
315 CALL assert(lnumg,'NS/nranks < 2...use smaller number of processors')
316 
317 mappingdone=.true.
318 END SUBROUTINE bcyclicmapping

325 SUBROUTINE computeallgatherparameters
326 
327 INTEGER :: i

337 END DO
338

340 DO i=2,activeranks
341  dispns(i)=dispns(i-1)+rcountsns(i-1)
342  disp(i)=disp(i-1)+rcounts(i-1)
343 END DO

352 
353 !INTEGER, DIMENSION(:), INTENT(IN) :: qarray
354 INTEGER, INTENT(IN) :: query
355 LOGICAL, INTENT(OUT) :: found
356 INTEGER, INTENT(OUT) :: location

363 IF (activeranks.GT.n) activeranks=n

369  found=.true.

400 SUBROUTINE send(columnData,blockRowNum,blockRowType,columnNum, procNum)
401 
402 REAL(dp), DIMENSION(1:M), INTENT(IN) :: columndata
403 INTEGER, INTENT(IN) :: blockrownum, blockrowtype, columnnum
404 INTEGER, INTENT(OUT) :: procnum
405 
406 CHARACTER, DIMENSION(PACKSIZE) :: sendbuf
407 INTEGER :: positn
408 LOGICAL :: found
409 
410 positn=0
411 CALL mpi_pack(blockrownum,1,mpi_integer,sendbuf,packsize,positn,siesta_comm,mpi_err)
412 CALL mpi_pack(columnnum,1,mpi_integer,sendbuf,packsize,positn,siesta_comm,mpi_err)
413 CALL mpi_pack(columndata,m,mpi_real8,sendbuf,packsize,positn,siesta_comm,mpi_err)
414 
415 found=.false.
416 CALL search(blockrownum,found,procnum)
417 IF (found) THEN
418  IF(procnum-1.EQ.rank) THEN
419  IF (blockrowtype.EQ.upper) THEN
420  CALL setblocktridatastruct(upper,blockrownum,columnnum,columndata)
421  ELSE IF (blockrowtype.EQ.savediag) THEN
422  CALL setblocktridatastruct(savediag,blockrownum,columnnum,columndata)
423  ELSE IF (blockrowtype.EQ.diag) THEN
424  CALL setblocktridatastruct(diag,blockrownum,columnnum,columndata)
425  ELSE IF (blockrowtype.EQ.lower) THEN
426  CALL setblocktridatastruct(lower,blockrownum,columnnum,columndata)
427  ELSE
428  CALL assert(.false.,'send error, blockRowType:')
429  END IF
430  ELSE
431  IF (buffered) THEN
432  CALL mpi_bsend(sendbuf,positn,mpi_packed,procnum-1,blockrowtype,siesta_comm,mpi_err)
433  ELSE
434  CALL mpi_send(sendbuf,positn,mpi_packed,procnum-1,blockrowtype,siesta_comm,mpi_err)
435  END IF
436  END IF
437 ELSE
438  CALL mpi_abort(mpi_comm_world,1,mpi_err)
439 END IF

441 END SUBROUTINE send

445 SUBROUTINE receive (IPROBEFLAG)

447 LOGICAL, INTENT(IN) :: iprobeflag

458  &mpi_any_tag,siesta_comm,mpi_status,mpi_err)
459  nrecd=nrecd+1
460 
461  it=mpi_status(mpi_tag)
462  IF (it.NE.1.AND.it.NE.2.AND.it.NE.3.AND.it.NE.4) THEN
463  WRITE(tofu,*) 'MPI_TAG:',it; FLUSH(tofu)
464  CALL assert(.false.,'receive 1 error')
465  END IF
466  positn=0
467  CALL mpi_unpack(recvbuf,packsize,positn,br,1,mpi_integer,siesta_comm,mpi_err)
468  CALL mpi_unpack(recvbuf,packsize,positn,ic,1,mpi_integer,siesta_comm,mpi_err)

476 
477  IF (it.EQ.1) THEN !UPPER DIAGONAL
478  CALL setblockrowcol(br,ic,coldata,1)
479  ELSE IF (it.EQ.2) THEN !MAIN DIAGONAL
480  CALL setblockrowcol(br,ic,coldata,2)
481  ELSE IF (it.EQ.3) THEN !LOWER DIAGONAL
482  CALL setblockrowcol(br,ic,coldata,3)
483  ELSE IF (it.EQ.4) THEN !SAVED DIAGONAL
484  CALL storediagonal(br,ic,coldata)
485  ELSE
486  WRITE(*,*)'Something wrong in ', rank,mpi_status(mpi_tag),br,ic,it
487  CALL assert(.false.,'receive 3 error')
488  END IF
489 END IF
490 
491 END SUBROUTINE receive
