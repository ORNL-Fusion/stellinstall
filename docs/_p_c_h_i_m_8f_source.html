45 ! PCHIM: Piecewise Cubic Hermite Interpolation to
46 ! Monotone data.
47 !
48 ! Sets derivatives needed to determine a monotone piecewise cubi!
49 ! Hermite interpolant to the data given in X and F.
50 !
51 ! Default boundary conditions are provided which are compatible
52 ! with monotonicity. (See PCHIC IF user control of boundary con-
53 ! ditions is desired.)
54 !
55 ! If the data are ONLY piecewise monotonic, the interpolant will
56 ! have an extremum at each point where monotonicity switches direc-
57 ! tion. (See PCHIC IF user control is desired in such cases.)
58 !
59 ! To facilitate two-dimensional applications, includes an increment
60 ! between successive values of the F- and D-arrays.
61 !
62 ! The resulting piecewise cubic Hermite FUNCTION may be evaluated
63 ! by PCHFE or PCHFD.
64 !
65 ! ----------------------------------------------------------------------
66 !
67 ! Calling sequence:
68 !
69 ! PARAMETER (INCFD = ...)
70 ! INTEGER N, IERR
71 ! REAL X(N), F(INCFD,N), D(INCFD,N)
72 !
73 ! CALL PCHIM (N, X, F, D, INCFD, IERR)
74 !
75 ! Parameters:
76 !
77 ! N -- (input) number of data points. (Error RETURN IF N.LT.2 .)
78 ! If N=2, simply does linear interpolation.
79 !
80 ! X -- (input) REAL array of independent variable values. The
81 ! elements of X must be strictly increasing:
82 ! X(I-1) .LT. X(I), I = 2(1)N.
83 ! (Error RETURN IF not.)
84 !
85 ! F -- (input) REAL array of dependent variable values to be inter-
86 ! polated. F(1+(I-1)*INCFD) is value corresponding to X(I).
87 ! PCHIM is designed for monotonic data, but it will work for
88 ! ANY F-array. It will force extrema at points WHERE mono-
89 ! tonicity switches direction. If some other treatment of
90 ! switch points is desired, PCHIC should be used instead.
91 ! -----
92 ! D -- (output) REAL array of derivative values at the data points.
93 ! If the data are monotonic, these values will determine a
94 ! a monotone cubic Hermite function.
95 ! The value corresponding to X(I) is stored in
96 ! D(1+(I-1)*INCFD), I=1(1)N.
97 ! No other entries in D are changed.
98 !
99 ! INCFD -- (input) increment between successive values in F and D.
100 ! This argument is provided primarily for 2-D applications.
101 ! (Error RETURN IF INCFD.LT.1 .)
102 !
103 ! IERR -- (output) error flag.
104 ! Normal RETURN:
105 ! IERR = 0 (no errors).
106 ! Warning error:
107 ! IERR.GT.0 means that IERR switches in the direction
108 ! of monotonicity were detected.
109 ! "Recoverable" errors:
110 ! IERR = -1 IF N.LT.2 .
111 ! IERR = -2 IF INCFD.LT.1 .
112 ! IERR = -3 IF the X-array is not strictly increasing.
113 ! (The D-array has not been changed in ANY of these cases.)
114 ! NOTE: The above errors are checked in the order listed,
115 ! and following arguments have **NOT** been validated.
116 !
117 !***REFERENCES 1. F.N.FRITSCH AND R.E.CARLSON, 'MONOTONE PIECEWISE CUBIC INTERPOLATION,'

158  DO i = 2, n
159  IF (x(i) <= x(i-1)) GO TO 5003
160  END DO

162 ! FUNCTION DEFINITION IS OK, GO ON.
163 !
164  ierr = 0
165  nless1 = n - 1
166  h1 = x(2) - x(1)
167  del1 = (f(1,2)-f(1,1))/h1
168  dsave = del1
169 !
170 ! SPECIAL CASE N=2 -- USE LINEAR INTERPOLATION.
171 !
172  IF (nless1 <= 1) THEN
173  d(1,1) = del1
174  d(1,n) = del1
175  ELSE
176 !
177 ! NORMAL CASE (N .GE. 3).
178 !
179  h2 = x(3) - x(2)
180  del2 = (f(1,3)-f(1,2))/h2
181 !
182 ! SET D(1) VIA NON-CENTERED THREE-POINT FORMULA, ADJUSTED TO BE
183 ! SHAPE-PRESERVING.
184 !
185  hsum = h1 + h2
186  w1 = (h1 + hsum)/hsum
187  w2 = -h1/hsum
188  d(1,1) = w1*del1 + w2*del2
189  IF (pchst(d(1,1),del1) <= zero) THEN
190  d(1,1) = zero
191  ELSE IF (pchst(del1,del2) < zero) THEN
192 ! NEED DO THIS CHECK ONLY IF MONOTONICITY SWITCHES.
193  d_max = three*del1
194  IF (abs(d(1,1)) > abs(d_max)) d(1,1) = d_max
195  ENDIF
196 !
197 ! LOOP THROUGH INTERIOR POINTS.
198 !

200  IF (i /= 2) THEN

207  ENDIF
208 !
209 ! SET D(I)=0 UNLESS DATA ARE STRICTLY MONOTONIC.
210 !
211  d(1,i) = zero
212  r1 = pchst(del1,del2)
213  IF (r1 >= zero) THEN
214  IF (r1 > zero) GO TO 45
215  IF (del2 == zero) cycle
216  IF (pchst(dsave,del2) < zero) ierr = ierr + 1
217  dsave = del2
218  cycle

220  ENDIF
221  ierr = ierr + 1
222  dsave = del2
223  cycle

226 !

228  hsumt3 = hsum + hsum + hsum

230  w2 = (hsum + h2)/hsumt3

232  d_min = min(abs(del1),abs(del2))
233  drat1 = del1/d_max
234  drat2 = del2/d_max
235  d(1,i) = d_min/(w1*drat1 + w2*drat2)
236 !
237  END DO

240 ! SHAPE-PRESERVING.
241 !

243  w2 = (h2 + hsum)/hsum

245  IF (pchst(d(1,n),del2) <= zero) THEN
246  d(1,n) = zero
247  ELSE IF (pchst(del1,del2) < zero) THEN
248 ! NEED DO THIS CHECK ONLY IF MONOTONICITY SWITCHES.
249  d_max = three*del2
250  IF (abs(d(1,n)) > abs(d_max)) d(1,n) = d_max
251  ENDIF
252 !
253 ! NORMAL RETURN.
254 !
255  ENDIF
256  RETURN
257 !
258 ! ERROR RETURNS.
259 !
260  ENDIF

264  RETURN

270  RETURN
271 !

274  ierr = -3
275  stop 'PCHIM -- X-ARRAY NOT STRICTLY INCREASING'
