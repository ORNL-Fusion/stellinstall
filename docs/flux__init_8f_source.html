1  SUBROUTINE flux_init(phipog)
2  USE vmec_main
3  USE vmec_params, ONLY: signgs
4  USE vforces, ONLY : r12=>armn_o, gsqrt=>azmn_o, orsq=>blmn_o
5  USE vsvd
6  USE realspace
7  IMPLICIT NONE
8 C-----------------------------------------------
9 C D u m m y A r g u m e n t s
10 C-----------------------------------------------
11  REAL(rprec), DIMENSION(*) :: phipog
12 C-----------------------------------------------
13 C L o c a l P a r a m e t e r s
14 C-----------------------------------------------
15  REAL(rprec), PARAMETER :: p5 = 0.5_dp, c1p5 = 1.5_dp
16 C-----------------------------------------------
17 C L o c a l V a r i a b l e s
18 C-----------------------------------------------
19  INTEGER :: l, js
20 C-----------------------------------------------
21 
22 !
23 ! COMPUTE OBSERVATION POINT - INVARIANT FUNCTIONS OF RADIUS
24 ! CURRENT = PHIP * INTEGRAL(0,2pi)[ guu / gsqrt] (on full mesh)
25 ! RM2 = < R**(-2) >
26 ! VRM2 = V` * RM2 (V` = 2pi * VP)
27 ! ORSQ = SQRT(G) * R**(-2) (on half mesh)
28 !
29 ! MUST HAVE GONE THROUGH NEWPROFILE DETERMINATION OF IOTA AT
30 ! LEAST ONCE, OTHERWISE IOTAS IS UNDEFINED!
31 !
32  IF (iresidue .le. 0) RETURN
33  current(1) = zero
34  presint(1) = one
35 
36  DO l = 2,nrzt-1
37  orsq(l) = p5*( phipog(l) + phipog(l+1) ) *
38  1 (ru0(l)*ru0(l) + zu0(l)*zu0(l))
39  ENDDO
40  DO l = ns,nrzt,ns
41  orsq(l) = ( c1p5*phipog(l) - p5*phipog(l-1) ) *
42  1 (ru0(l)*ru0(l) + zu0(l)*zu0(l))
43  ENDDO
44 
45  DO js = 2, ns
46  current(js) = twopi*sum(orsq(js:nrzt:ns)*wint(js:nrzt:ns))
47  presint(js) = one
48  END DO
49 
50  DO l = 2, nrzt
51  orsq(l) = gsqrt(l)/r12(l)**2
52  END DO
53  DO js = 2, ns
54  vrm2(js) = twopi*sum(orsq(js:nrzt:ns)*wint(js:nrzt:ns))
55  rm2(js) = vrm2(js)/(twopi*signgs*vp(js))
56  ovrm2(js) = one/vrm2(js)
57  ochip(js) = one/(phip(js)*iotas(js))
58  presph(js) = presf(js) - presf(js - 1)
59  END DO
60 
61  END SUBROUTINE flux_init
