10  REAL(rprec), DIMENSION(N) :: x

12  REAL(rprec), DIMENSION(2,N) :: wk

14 ! L o c a l V a r i a b l e s
15 !-----------------------------------------------
16  INTEGER :: ibeg, iend, index, j, nm1
17  REAL(rprec) :: g, half, one
18  REAL(rprec), DIMENSION(3) :: stemp
19  REAL(rprec) :: three, two
20  REAL(rprec), DIMENSION(4) :: xtemp
21  REAL(rprec) :: zero

23 ! E x t e r n a l F u n c t i o n s

27 !***BEGIN PROLOGUE PCHSP
28 !***DATE WRITTEN 820503 (YYMMDD)
29 !***REVISION DATE 870707 (YYMMDD)
30 !***CATEGORY NO. E1B
31 !***KEYWORDS LIBRARY=SLATEC(PCHIP),
32 ! TYPE=SINGLE PRECISION(PCHSP-S DPCHSP-D),
33 ! CUBIC HERMITE INTERPOLATION,PIECEWISE CUBIC INTERPOLATION,
34 ! SPLINE INTERPOLATION
35 !***AUTHOR FRITSCH, F. N., (LLNL)
36 ! MATHEMATICS AND STATISTICS DIVISION
37 ! LAWRENCE LIVERMORE NATIONAL LABORATORY
38 ! P.O. BOX 808 (L-316)
39 ! LIVERMORE, CA 94550
40 ! FTS 532-4275, (415) 422-4275
41 !***PURPOSE Set derivatives needed to determine the Hermite represen-
42 ! tation of the cubic spline interpolant to given data, with
43 ! specified boundary conditions.
44 !***DESCRIPTION
45 !
46 ! PCHSP: Piecewise Cubic Hermite Spline
47 !
48 ! Computes the Hermite representation of the cubic spline inter-
49 ! polant to the data given in X and F satisfying the boundary
50 ! conditions specified by IC and VC.
51 !
52 ! To facilitate two-dimensional applications, includes an increment
53 ! between successive values of the F- and D-arrays.
54 !
55 ! The resulting piecewise cubic Hermite FUNCTION may be evaluated
56 ! by PCHFE or PCHFD.
57 !
58 ! NOTE: This is a modified version of C. de Boor''S cubic spline
59 ! routine CUBSPL.
60 !
61 ! ----------------------------------------------------------------------
62 !
63 ! Calling sequence:
64 !
65 ! PARAMETER (INCFD = ...)
66 ! INTEGER IC(2), N, NWK, IERR
67 ! REAL VC(2), X(N), F(INCFD,N), D(INCFD,N), WK(NWK)
68 !
69 ! CALL PCHSP (IC, VC, N, X, F, D, INCFD, WK, NWK, IERR)
70 !
71 ! Parameters:
72 !
73 ! IC -- (input) INTEGER array of length 2 specifying desired
74 ! boundary conditions:
75 ! IC(1) = IBEG, desired condition at beginning of data.
76 ! IC(2) = IEND, desired condition at END of data.
77 !
78 ! IBEG = 0 to set D(1) so that the third derivative is con-
79 ! tinuous at X(2). This is the "not a knot" condition
80 ! provided by de Boor''S cubic spline routine CUBSPL.
81 ! < This is the DEFAULT boundary condition. >
82 ! IBEG = 1 IF first derivative at X(1) is given in VC(1).
83 ! IBEG = 2 IF second derivative at X(1) is given in VC(1).
84 ! IBEG = 3 to USE the 3-point difference formula for D(1).
85 ! (Reverts to the DEFAULT b.c. IF N.LT.3 .)
86 ! IBEG = 4 to USE the 4-point difference formula for D(1).
87 ! (Reverts to the DEFAULT b.c. IF N.LT.4 .)
88 ! NOTES:
89 ! 1. An error RETURN is taken IF IBEG is out of range.
90 ! 2. For the "natural" boundary condition, USE IBEG=2 and
91 ! VC(1)=0.
92 !
93 ! IEND may take on the same values as IBEG, but applied to
94 ! derivative at X(N). In CASE IEND = 1 or 2, the value is
95 ! given in VC(2).
96 !
97 ! NOTES:
98 ! 1. An error RETURN is taken IF IEND is out of range.
99 ! 2. For the "natural" boundary condition, USE IEND=2 and
100 ! VC(2)=0.
101 !
102 ! VC -- (input) REAL array of length 2 specifying desired boundary
103 ! values, as indicated above.
104 ! VC(1) need be set ONLY IF IC(1) = 1 or 2 .
105 ! VC(2) need be set ONLY IF IC(2) = 1 or 2 .
106 !
107 ! N -- (input) number of data points. (Error RETURN IF N.LT.2 .)
108 !
109 ! X -- (input) REAL array of independent variable values. The
110 ! elements of X must be strictly increasing:
111 ! X(I-1) .LT. X(I), I = 2(1)N.
112 ! (Error RETURN IF not.)
113 !
114 ! F -- (input) REAL array of dependent variable values to be inter-
115 ! polated. F(1+(I-1)*INCFD) is value corresponding to X(I).
116 !
117 ! D -- (output) REAL array of derivative values at the data points.
118 ! These values will determine the cubic spline interpolant
119 ! with the requested boundary conditions.
120 ! The value corresponding to X(I) is stored in
121 ! D(1+(I-1)*INCFD), I=1(1)N.
122 ! No other entries in D are changed.
123 !
124 ! INCFD -- (input) increment between successive values in F and D.
125 ! This argument is provided primarily for 2-D applications.
126 ! (Error RETURN IF INCFD.LT.1 .)
127 !
128 ! WK -- (scratch) REAL array of working storage.
129 !
130 ! NWK -- (input) length of work array.
131 ! (Error RETURN IF NWK.LT.2*N .)
132 !
133 ! IERR -- (output) error flag.
134 ! Normal RETURN:
135 ! IERR = 0 (no errors).
136 ! "Recoverable" errors:
137 ! IERR = -1 IF N.LT.2 .
138 ! IERR = -2 IF INCFD.LT.1 .
139 ! IERR = -3 IF the X-array is not strictly increasing.
140 ! IERR = -4 IF IBEG.LT.0 or IBEG.GT.4 .
141 ! IERR = -5 IF IEND.LT.0 of IEND.GT.4 .
142 ! IERR = -6 IF both of the above are true.
143 ! IERR = -7 IF NWK is too small.
144 ! NOTE: The above errors are checked in the order listed,
145 ! and following arguments have **NOT** been validated.
146 ! (the d-array has not been changed in any of these cases.)
147 ! IERR = -8 in case of trouble solving the linear system
148 ! for the interior derivative values.
149 ! (The D-array may have been changed in this case.)
150 ! ( Do **NOT** USE it! )
151 !
152 !***REFERENCES CARL DE BOOR, A PRACTICAL GUIDE TO SPLINES, SPRINGER-
153 ! VERLAG (NEW YORK, 1978), PP. 53-59.
154 !***ROUTINES CALLED PCHDF
155 !***END PROLOGUE PCHSP
156 !
157 ! ----------------------------------------------------------------------
158 !
159 ! Change record:
160 ! 82-08-04 Converted to SLATEC library version.
161 ! 87-07-07 Minor cosmetic changes to prologue.
162 !
163 ! ----------------------------------------------------------------------
164 !
165 ! Programming notes:
166 !
167 ! To produce a Double precision version, simply:
168 ! a. Change PCHSP to DPCHSP wherever it occurs,
169 ! b. Change the real declarations to double precision, and
170 ! c. Change the constants ZERO, HALF, ... to Double precision.
171 !
172 ! DECLARE ARGUMENTS.
173 !
174 !
175 ! DECLARE LOCAL VARIABLES.
176 !
177 !
178  DATA zero/0/
179  DATA half/0.5_dp/
180  DATA one/1/
181  DATA two/2/
182  DATA three/3/
183 !
184 ! VALIDITY-CHECK ARGUMENTS.
185 !
186 !***FIRST EXECUTABLE STATEMENT PCHSP
187  IF (n >= 2) THEN
188  IF (incfd < 1) GO TO 5002
189  DO j = 2, n
190  IF (x(j) <= x(j-1)) GO TO 5003
191  END DO
192 !
193  ibeg = ic(1)
194  iend = ic(2)
195  ierr = 0
196  IF (ibeg<0 .OR. ibeg>4) ierr = ierr - 1
197  IF (iend<0 .OR. iend>4) ierr = ierr - 2
198  IF (ierr < 0) GO TO 5004
199 !
200 ! FUNCTION DEFINITION IS OK -- GO ON.
201 !
202  IF (nwk < 2*n) GO TO 5007
203 !
204 ! COMPUTE FIRST DIFFERENCES OF X SEQUENCE AND STORE IN WK(1,.). ALSO,
205 ! COMPUTE FIRST DIVIDED DIFFERENCE OF DATA AND STORE IN WK(2,.).
206  wk(1,2:n) = x(2:n) - x(:n-1)
207  wk(2,2:n) = (f(1,2:n)-f(1,:n-1))/wk(1,2:n)
208 !
209 ! SET TO DEFAULT BOUNDARY CONDITIONS IF N IS TOO SMALL.
210 !
211  IF (ibeg > n) ibeg = 0
212  IF (iend > n) iend = 0
213 !
214 ! SET UP FOR BOUNDARY CONDITIONS.
215 !
216  IF (ibeg==1 .OR. ibeg==2) THEN
217  d(1,1) = vc(1)
218  ELSE IF (ibeg > 2) THEN
219 ! PICK UP FIRST IBEG POINTS, IN REVERSE ORDER.
220  DO j = 1, ibeg
221  index = ibeg - j + 1
222 ! INDEX RUNS FROM IBEG DOWN TO 1.
223  xtemp(j) = x(index)
224  IF (j < ibeg) stemp(j) = wk(2,index)
225  END DO
226 ! --------------------------------
227  d(1,1) = pchdf(ibeg,xtemp,stemp,ierr)
228 ! --------------------------------
229  IF (ierr /= 0) GO TO 5009
230  ibeg = 1
231  ENDIF
232 !
233  IF (iend==1 .OR. iend==2) THEN
234  d(1,n) = vc(2)
235  ELSE IF (iend > 2) THEN
236 ! PICK UP LAST IEND POINTS.
237  DO j = 1, iend
238  index = n - iend + j
239 ! INDEX RUNS FROM N+1-IEND UP TO N.
240  xtemp(j) = x(index)
241  IF (j < iend) stemp(j) = wk(2,index+1)
242  END DO
243 ! --------------------------------
244  d(1,n) = pchdf(iend,xtemp,stemp,ierr)
245 ! --------------------------------
246  IF (ierr /= 0) GO TO 5009
247  iend = 1
248  ENDIF
249 !
250 ! --------------------( BEGIN CODING FROM CUBSPL )--------------------
251 !
252 ! **** A TRIDIAGONAL LINEAR SYSTEM FOR THE UNKNOWN SLOPES S(J) OF
253 ! F AT X(J), J=1,...,N, IS GENERATED AND THEN SOLVED BY GAUSS ELIM-
254 ! INATION, WITH S(J) ENDING UP IN D(1,J), ALL J.
255 ! WK(1,.) AND WK(2,.) ARE USED FOR TEMPORARY STORAGE.
256 !
257 ! CONSTRUCT FIRST EQUATION FROM FIRST BOUNDARY CONDITION, OF THE FORM
258 ! WK(2,1)*S(1) + WK(1,1)*S(2) = D(1,1)
259 !
260  IF (ibeg == 0) THEN
261  IF (n == 2) THEN
262 ! NO CONDITION AT LEFT END AND N = 2.
263  wk(2,1) = one
264  wk(1,1) = one
265  d(1,1) = two*wk(2,2)
266  ELSE
267 ! NOT-A-KNOT CONDITION AT LEFT END AND N .GT. 2.
268  wk(2,1) = wk(1,3)
269  wk(1,1) = wk(1,2) + wk(1,3)
270  d(1,1) = ((wk(1,2)+two*wk(1,1))*wk(2,2)
271  1 *wk(1,3)+wk(1,2)**2*wk(2,3))/wk(1,1)
272  ENDIF
273  ELSE IF (ibeg == 1) THEN
274 ! SLOPE PRESCRIBED AT LEFT END.
275  wk(2,1) = one
276  wk(1,1) = zero
277  ELSE
278 ! SECOND DERIVATIVE PRESCRIBED AT LEFT END.
279  wk(2,1) = two
280  wk(1,1) = one
281  d(1,1) = three*wk(2,2) - half*wk(1,2)*d(1,1)
282  ENDIF
283 !
284 ! IF THERE ARE INTERIOR KNOTS, GENERATE THE CORRESPONDING EQUATIONS AND
285 ! CARRY OUT THE FORWARD PASS OF GAUSS ELIMINATION, AFTER WHICH THE J-TH
286 ! EQUATION READS WK(2,J)*S(J) + WK(1,J)*S(J+1) = D(1,J).
287 !
288  nm1 = n - 1
289  IF (nm1 > 1) THEN
290  DO j = 2, nm1
291  IF (wk(2,j-1) == zero) GO TO 5008
292  g = -wk(1,j+1)/wk(2,j-1)
293  d(1,j) = g*d(1,j-1) + three*(wk(1,j)
294  1 *wk(2,j+1)+wk(1,j+1)*wk(2,j))
295  wk(2,j) = g*wk(1,j-1) + two*(wk(1,j)+wk(1,j+1))
296  END DO
297  ENDIF
298 !
299 ! CONSTRUCT LAST EQUATION FROM SECOND BOUNDARY CONDITION, OF THE FORM
300 ! (-G*WK(2,N-1))*S(N-1) + WK(2,N)*S(N) = D(1,N)
301 !
302 ! IF SLOPE IS PRESCRIBED AT RIGHT END, ONE CAN GO DIRECTLY TO BACK-
303 ! SUBSTITUTION, SINCE ARRAYS HAPPEN TO BE SET UP JUST RIGHT FOR IT
304 ! AT THIS POINT.
305  IF (iend /= 1) THEN
306 !
307  IF (iend == 0) THEN
308  IF (n==2 .AND. ibeg==0) THEN
309 ! NOT-A-KNOT AT RIGHT ENDPOINT AND AT LEFT ENDPOINT AND N = 2.
310  d(1,2) = wk(2,2)
311  GO TO 30
312  ELSE IF (n==2 .OR. n==3 .AND. ibeg==0) THEN
313 ! EITHER (N=3 AND NOT-A-KNOT ALSO AT LEFT) OR (N=2 AND *NOT*
314 ! NOT-A-KNOT AT LEFT END POINT).
315  d(1,n) = two*wk(2,n)
316  wk(2,n) = one
317  IF (wk(2,n-1) == zero) GO TO 5008
318  g = -one/wk(2,n-1)
319  ELSE
320 ! NOT-A-KNOT AND N .GE. 3, AND EITHER N.GT.3 OR ALSO NOT-A-
321 ! KNOT AT LEFT END POINT.
322  g = wk(1,n-1) + wk(1,n)
323 ! DO NOT NEED TO CHECK FOLLOWING DENOMINATORS (X-DIFFERENCES).
324  d(1,n) = ((wk(1,n)+two*g)*wk(2,n)*wk(1,n-1)
325  1 +wk(1,n)**2*(f(1,n-1)-f(1,n-2))/wk(1,n-1))/g
326  IF (wk(2,n-1) == zero) GO TO 5008
327  g = -g/wk(2,n-1)
328  wk(2,n) = wk(1,n-1)
329  ENDIF
330  ELSE
331 ! SECOND DERIVATIVE PRESCRIBED AT RIGHT ENDPOINT.
332  d(1,n) = three*wk(2,n) + half*wk(1,n)*d(1,n)
333  wk(2,n) = two
334  IF (wk(2,n-1) == zero) GO TO 5008
335  g = -one/wk(2,n-1)
336  ENDIF
337 !
338 ! COMPLETE FORWARD PASS OF GAUSS ELIMINATION.
339 !
340  wk(2,n) = g*wk(1,n-1) + wk(2,n)
341  IF (wk(2,n) == zero) GO TO 5008
342  d(1,n) = (g*d(1,n-1)+d(1,n))/wk(2,n)
343 !
344 ! CARRY OUT BACK SUBSTITUTION
345 !
346  ENDIF
347  30 CONTINUE
348  DO j = nm1, 1, -1
349  IF (wk(2,j) == zero) GO TO 5008
350  d(1,j) = (d(1,j)-wk(1,j)*d(1,j+1))/wk(2,j)
351  END DO
352 ! --------------------( END CODING FROM CUBSPL )--------------------
353 !
354 ! NORMAL RETURN.
355 !
356  RETURN
357 !
358 ! ERROR RETURNS.
359 !
360  ENDIF
361 ! N.LT.2 RETURN.
362  ierr = -1
363  stop 'PCHSP -- NUMBER OF DATA POINTS LESS THAN TWO'
364  RETURN
365 !
366  5002 CONTINUE
367 ! INCFD.LT.1 RETURN.
368  ierr = -2
369  stop 'PCHSP -- INCREMENT LESS THAN ONE'
370  RETURN
371 !
372  5003 CONTINUE
373 ! X-ARRAY NOT STRICTLY INCREASING.
374  ierr = -3
375  stop 'PCHSP -- X-ARRAY NOT STRICTLY INCREASING'
376  RETURN
377 !
378  5004 CONTINUE
379 ! IC OUT OF RANGE RETURN.
380  ierr = ierr - 3
381  stop 'PCHSP -- IC OUT OF RANGE'
382  RETURN
383 !
384  5007 CONTINUE
385 ! NWK TOO SMALL RETURN.
386  ierr = -7
387  stop 'PCHSP -- WORK ARRAY TOO SMALL'
388  RETURN
389 !
390  5008 CONTINUE
391 ! SINGULAR SYSTEM.
392 ! *** THEORETICALLY, THIS CAN ONLY OCCUR IF SUCCESSIVE X-VALUES ***
393 ! *** ARE EQUAL, WHICH SHOULD ALREADY HAVE BEEN CAUGHT (IERR=-3). ***
394  ierr = -8
395  stop 'PCHSP -- SINGULAR LINEAR SYSTEM'
396  RETURN
397 !
398  5009 CONTINUE
399 ! ERROR RETURN FROM PCHDF.
400 ! *** THIS CASE SHOULD NEVER OCCUR ***
401  ierr = -9
402  stop 'PCHSP -- ERROR RETURN FROM PCHDF'
403  RETURN
404 !------------- LAST LINE OF PCHSP FOLLOWS ------------------------------
405  END SUBROUTINE pchsp
