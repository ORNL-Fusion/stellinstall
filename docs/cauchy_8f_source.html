1  SUBROUTINE cauchy(rbdy, zbdy, rubdy, zubdy, rlim, zlim, residue,
2  1 sep, distmax, ntheta, nlim)
3  USE vparams, ONLY: twopi, dp, rprec
4  IMPLICIT NONE
5 C-----------------------------------------------
6 C D u m m y A r g u m e n t s
7 C-----------------------------------------------
8  INTEGER, INTENT(in) :: ntheta, nlim
9  REAL(rprec), INTENT(in) :: distmax
10  REAL(rprec), DIMENSION(ntheta), INTENT(in) ::
11  1 rbdy, zbdy, rubdy, zubdy
12  REAL(rprec), DIMENSION(nlim) :: rlim, zlim, residue, sep
13 C-----------------------------------------------
14 C L o c a l P a r a m e t e r s
15 C-----------------------------------------------
16  REAL(rprec), PARAMETER :: zero=0, p5=0.5_dp, two=2
17 C-----------------------------------------------
18 C L o c a l V a r i a b l e s
19 C-----------------------------------------------
20  INTEGER :: n, i, imin, imax
21  REAL(rprec), DIMENSION(ntheta) :: dsq, dsepdu
22  REAL(rprec), DIMENSION(ntheta) :: x1u, y1u
23  REAL(rprec) :: delu, dmin, delta_d, alpha, gam0
24 C-----------------------------------------------
25 c Check that the points (rlim(i),zlim(i)) are inside boundary surface
26 c using Cauchy's theorem in "complex"-plane (for a fixed
27 c toroidal plane, nphi=const. It is assumed that rbdy, zbdy are
28 c extended around the full interval, 0-2pi, in theta.
29 c with rbdy(1) = rbdy(ntheta+1) (i.e., ntheta intervals)
30 c
31 c Because of numerical inaccuracies, instead of testing on
32 c res = 0 (outside), res = 1 (inside), we use the test:
33 c res >= .5, inside; res < .5, outside
34 c**********************************************************************
35 c
36 c LOCAL VARIABLE ARRAYS
37 c
38 c dsq: Distance squared between limiter point and plasma boundary
39 c sep: Minimum dsq for each limiter point
40 c dsepdu: .5* d(dsq)/d(theta)
41 c residue: Contour integral of 1/(X-rlim)in complex X=(R,Z) plane
42 c
43  delu = twopi/ntheta
44  dmin = 1.e-20_dp*distmax
45 
46  DO n = 1, nlim
47  residue(n) = zero
48  x1u = rbdy(:ntheta) - rlim(n)
49  y1u = zbdy(:ntheta) - zlim(n)
50  dsq(:ntheta) = x1u*x1u + y1u*y1u
51  dsepdu(:ntheta) = x1u*rubdy(:ntheta) + y1u*zubdy(:ntheta)
52  residue(n) = residue(n) + sum((x1u*zubdy(:ntheta) -
53  1 y1u*rubdy(:ntheta))/(dsq(:ntheta)+dmin))
54 
55  residue(n) = residue(n)/ntheta
56 !
57 ! Find actual minimum distance from nth limiter point to boundary
58 !
59  sep(n) = distmax
60  DO i = 1,ntheta
61  IF( dsq(i).le.sep(n) )then
62  imin = i
63  sep(n) = dsq(i)
64  ENDIF
65  ENDDO
66 
67 ! gamu = two*ABS(dsepdu(imin))*delu
68 
69  IF (dsepdu(imin) .le. zero) THEN
70  imax = 1 + mod(imin,ntheta)
71  ELSE
72  imax = imin
73  imin = imax - 1
74  IF (imin .eq. 0) imin = ntheta
75  ENDIF
76 
77  delta_d = two*(dsepdu(imax)-dsepdu(imin))
78  alpha = delta_d/delu
79 ! gamu = gamu/delta_d
80 ! sep(n) = sep(n) - p5*alpha*gamu**2
81  IF (alpha .ne. zero)
82  1 gam0 = 0.5_dp - (dsq(imax)-dsq(imin))/(alpha*delu**2)
83  sep(n) = dsq(imin) - p5*alpha*(gam0*delu)**2
84  IF (sep(n) .lt. zero) sep(n) = zero
85 
86  END DO
87 
88  END SUBROUTINE cauchy
