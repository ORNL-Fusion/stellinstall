1  SUBROUTINE splinint(grn, cm, jacob, h, u, u1, w, w1, nk, nots,
2  1 ifunc, nmesh)
3  USE vparams
4  USE vsvd
5  USE vspline
6  IMPLICIT NONE
7 C-----------------------------------------------
8 C D u m m y A r g u m e n t s
9 C-----------------------------------------------
10  INTEGER nots, ifunc, nmesh
11  INTEGER, DIMENSION(*) :: nk
12  REAL(rprec), DIMENSION(nmesh) :: grn, cm
13  REAL(rprec), DIMENSION(nots) :: jacob, h
14  REAL(rprec), DIMENSION(nmesh) :: u, u1, w, w1
15 C-----------------------------------------------
16 C L o c a l V a r i a b l e s
17 C-----------------------------------------------
18  INTEGER :: j, k, nb, ia, ib, k1, ksp1, nmesh1
19  REAL(rprec), DIMENSION(nmesh) :: func
20  REAL(rprec), DIMENSION(nots) :: af, bs
21 C-----------------------------------------------
22 
23 !
24 ! COMPUTES af,bs FACTORS IN
25 ! (ifunc=INTFUN) Int[ GRN(X) * F(X) ds ] or
26 ! (ifunc=INTDER) Int[ GRN(X) * {d(CmF)/ds} ds ]
27 ! = af(k)*f(k) + bs(k)*f''(k)
28 ! WHERE f(k),f''(k) ARE SPLINE COEFFICIENTS OF F(X)
29 !
30 ! NOTE: FOR ifunc = INTDER, the OHS factor in CmF cancels
31 ! THE HS factor in the Integral.
32 ! FOR ifunc = INTFUN, GRN is assumed to be pre-multiplied
33 ! OUTSIDE this routine by HS factor
34 ! ALSO, COMPUTES af(k) + (SUM on i)bs(i)*J(i,k) = jacob(k),
35 ! WHERE J(i,k) = d[g(i)]/d[f(k)], g = f''
36 !
37 ! nk(k): Number of smesh-pts in k-th spline interval
38 ! xknots(k) < smesh <= xknots(k+1), k = 1,nots-1
39 !
40 ! NOTE: The ifunc=INTDER CASE is done by integrating by parts,
41 ! so that the half-point integration (GRN at half mesh pts)
42 ! becomes a full-point integration in Cm*F.
43 !
44  nb = ideriv !Pressure, iota derivatives vanish at origin
45  ksp1 = nots - 1
46  nmesh1 = nmesh - 1
47 
48  IF (ifunc .eq. intder) THEN
49 !
50 ! Integrate by parts (in s), func(1) and func(nmesh) are 'surface terms'
51 !
52  func(1) = -cm(1)*grn(2)
53  func(2:nmesh1) = cm(2:nmesh1)*(grn(2:nmesh1)-grn(3:nmesh1+1))
54  func(nmesh) = cm(nmesh)*grn(nmesh)
55  ELSE
56  func = grn
57  ENDIF
58 
59  af(:nots) = zero
60  bs(:nots) = zero
61 
62  ia = 1
63  DO k = 1, ksp1
64  IF (nk(k) .ne. 0) THEN
65  k1 = k + 1
66  ib = ia + nk(k) - 1
67  DO j = ia,ib
68  af(k) = af(k) + func(j)*w(j)
69  bs(k) = bs(k) + func(j)*u(j)
70  af(k1) = af(k1) + func(j)*w1(j)
71  bs(k1) = bs(k1) + func(j)*u1(j)
72  ENDDO
73  ia = ib + 1
74  ENDIF
75  END DO
76 
77  IF (ib .ne. nmesh) stop 'ib!=nmesh'
78  IF (nb .eq. natur) bs(1) = 0. !Natural boundary conditions
79  bs(nots) = 0.
80  CALL jacprod (bs, h, nots, nb) !Returns bs(i)=bs(j)*J(j,i)
81  jacob(:nots) = af(:nots) + bs(:nots)
82 
83  END SUBROUTINE splinint
