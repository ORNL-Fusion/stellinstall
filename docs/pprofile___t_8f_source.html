285  FUNCTION pprofile_get_value(this, s_arg)
286  USE line_segment
287  USE functions
288 
289  IMPLICIT NONE
290 
291 ! Declare Arguments
292  REAL(rprec) :: pprofile_get_value
293  type(pprofile_class), INTENT (in) :: this
294  REAL(rprec), INTENT(in) :: s_arg
295 
296 ! local variables
297  REAL(rprec) :: s_use
298  REAL(rprec) :: s_01
299  LOGICAL :: l_01
300  INTEGER :: i, iflag
301  REAL (rprec) :: start_time
302 
303 ! Start of executable code
304  start_time = profiler_get_start_time()
305 
306 ! Define argument and logical variable for s between zero and one
307  s_01 = max(zero, min(one,s_arg))
308  l_01 = (s_arg .ge. zero) .and. (s_arg .le. one)
309 
310  SELECT CASE(this%p_type)
311 
312  CASE (pprofile_none_type)
313  pprofile_get_value = -1.e49_rprec
314 
315 ! v3fit uses an extra varible as an offset than vmec as a result the b array is
316 ! one index shorter than what two_power takes.
317  CASE (pprofile_two_power_type)
318  pprofile_get_value = this%b(0)
319  IF (l_01) THEN
320  pprofile_get_value = pprofile_get_value &
321  & + two_power(s_arg, this%b(1:iub_b))
322  ENDIF
323 
324 ! v3fit uses an extra varible as an offset than vmec as a result the b array is
325 ! one index shorter than what two_power takes. This profile reverses the s
326 ! argument so that the maximum amplitude is at s=1 and the minimum is at s = 0.
327  CASE (pprofile_two_power_r_type)
328  pprofile_get_value = this%b(0)
329  IF (l_01) THEN
330  s_use = 1.0 - s_arg
331  pprofile_get_value = pprofile_get_value &
332  & + two_power(s_use, this%b(1:iub_b))
333  ENDIF
334 
335 ! v3fit uses an extra varible as an offset than vmec as a result the b array is
336 ! one index shorter than what two_power_gs takes. As a result only 5 guassian
337 ! peaks maybe specified instead of the 6 vmec can use.
338  CASE (pprofile_two_power_gs_type)
339  pprofile_get_value = this%b(0)
340  IF (l_01) THEN
341  pprofile_get_value = pprofile_get_value &
342  & + two_power_gs(s_arg, &
343  & this%b(1:iub_b))
344  ENDIF
345 
346  CASE (pprofile_power_series_type)
347  pprofile_get_value = zero
348  DO i = iub_b, ilb_b, -1 ! all of b array, backwards
349  pprofile_get_value = pprofile_get_value*s_01 &
350  & + this%b(i)
351  END DO
352 
353  CASE (pprofile_cubic_spline_type)
354  s_use = min(this%as(this%maxSplineIndex), &
355  & max(s_arg, this%as(1)))
356  CALL spline_cubic(s_use, pprofile_get_value, &
357  & this%as, this%af, &
358  & this%maxSplineIndex, iflag)
359  IF (iflag .ge. 0) THEN
360 
361  CALL profiler_set_stop_time('pprofile_get_value', &
362  & start_time)
363 
364  RETURN
365  ELSE IF (iflag .eq. -1) THEN
366  WRITE (*,*) 'ERROR: pprofile: outside value from ' // &
367  & 'spline_cubic'
368  ELSE IF(iflag .eq. -2) THEN
369  WRITE (*,*) 'ERROR: pprofile: decreasing s values ' // &
370  & 'in spline_cubic'
371  ELSE
372  WRITE (*,*) 'ERROR: pprofile: unknown error from ' // &
373  & 'spline_cubic'
374  ENDIF
375 ! Splines shuld not reach this point without triggering an error.
376  CALL exit(1)
377 
378  CASE (pprofile_akima_spline_type)
379  s_use = min(this%as(this%maxSplineIndex), &
380  & max(s_arg,this%as(1)))
381  CALL spline_akima(s_use, pprofile_get_value, &
382  & this%as, this%af, &
383  & this%maxSplineIndex, iflag)
384  IF (iflag < 0) THEN
385  WRITE (*,*) 'ERROR: pprofile: bad value from ' // &
386  & 'spline_akima requested'
387  CALL exit(1)
388  ENDIF
389 
390  CASE (pprofile_line_segment_type)
391  CALL line_seg(s_arg, pprofile_get_value, this%as, this%af, &
392  & this%maxSplineIndex)
393 
394  END SELECT
395 
396  CALL profiler_set_stop_time('pprofile_get_value', start_time)
397 
398  END FUNCTION
