94

135  n1=max(1,startglobrow); n2=min(ns,endglobrow)
136  IF (n1 .EQ. 1) pfilter(:,:,1) = 0
137  stes(1) = sum((pijh0(:,:,n1:n2)-pfilter(:,:,n1:n2))**2)
138  stes(2) = sum((pijh0(:,:,n1:n2)+pfilter(:,:,n1:n2))**2)
139 #if defined(MPI_OPT)
140  CALL mpi_allreduce(stes, rtes, 2, mpi_real8, mpi_sum, &
141  siesta_comm, mpi_err)
142  IF (rtes(2) .NE. zero) ste(1) = sqrt(rtes(1)/rtes(2))
143 #else
144  IF (stes(2) .NE. zero) ste(1) = sqrt(stes(1)/stes(2))
145 #endif
146  DEALLOCATE (pfilter)
147 
148  CALL checkfft(jbsupsmnsh, jbsupumnch, jbsupvmnch, jpmnch, isym)
149  IF (lasym) &
150  CALL checkfft(jbsupsmnch, jbsupumnsh, jbsupvmnsh, jpmnsh, iasym)
151 
152  IF (.NOT.lpar) THEN
153  startglobrow=startr; endglobrow=endr
154  END IF
155 
156  END SUBROUTINE init_state

363  fourv)

400  END IF
401

414  CALL toijsp_par(pmnh, pijh0_dv, nfour, four)
415
