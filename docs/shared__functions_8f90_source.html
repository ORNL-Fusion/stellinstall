106  SUBROUTINE getfsq(fsqout)
107  USE quantities, ONLY: fsubsmncf, fsubumnsf, fsubvmnsf, &
108  fsupsmncf, fsupumnsf, fsupvmnsf, &
109  fsubsmnsf, fsubumncf, fsubvmncf, &
110  fsupsmnsf, fsupumncf, fsupvmncf, &

113 ! D u m m y A r g u m e n t s

116 !-----------------------------------------------
117 ! L o c a l V a r i a b l e s
118 !-----------------------------------------------
119  REAL(dp) :: stmp(3), rtmp(3)
120  INTEGER :: js
121 !-----------------------------------------------
122  stmp(1) = sum(fsubsmncf(:,:,nsmin:nsmax)**2)
123  stmp(2) = sum(fsubumnsf(:,:,nsmin:nsmax)**2)
124  stmp(3) = sum(fsubvmnsf(:,:,nsmin:nsmax)**2)
125  IF (lasym) THEN
126  stmp(1) = stmp(1) + sum(fsubsmnsf(:,:,nsmin:nsmax)**2)
127  stmp(2) = stmp(2) + sum(fsubumncf(:,:,nsmin:nsmax)**2)
128  stmp(3) = stmp(3) + sum(fsubvmncf(:,:,nsmin:nsmax)**2)
129  END IF
130 
131 #if defined(MPI_OPT)
132  IF (parsolver) THEN
133  CALL mpi_allreduce(stmp,rtmp,3,mpi_real8,mpi_sum, &
134  siesta_comm,mpi_err)
135  stmp = rtmp
136  END IF
137 #endif
138  fsqvs=hs*stmp(1); fsqvu=hs*stmp(2); fsqvv=hs*stmp(3)
139 
140  CALL toupper_forces
141 
142 !VOLUME-AVERAGE |F|**2
143  stmp(1) = 0
144  DO js = nsmin,nsmax
145  stmp(1) = stmp(1) + vp_f(js)*sum( &
146  fsupsmncf(:,:,js)*fsubsmncf(:,:,js) &
147  + fsupumnsf(:,:,js)*fsubumnsf(:,:,js) &
148  + fsupvmnsf(:,:,js)*fsubvmnsf(:,:,js))
149  IF (lasym) THEN
150  stmp(1) = stmp(1) + vp_f(js)*sum( &
151  fsupsmnsf(:,:,js)*fsubsmnsf(:,:,js) &
152  + fsupumncf(:,:,js)*fsubumncf(:,:,js) &
153  + fsupvmncf(:,:,js)*fsubvmncf(:,:,js))
154  END IF
155  END DO
156  stmp(2) = sum(vp_f(nsmin:nsmax))
157 #if defined(MPI_OPT)
158  IF (parsolver) THEN
159  CALL mpi_allreduce(stmp,rtmp,2,mpi_real8,mpi_sum, &
160  siesta_comm,mpi_err)
161  stmp = rtmp
162  END IF
163 #endif
164  fsqout = stmp(1)/stmp(2)
165 
166 
167  END SUBROUTINE getfsq
