37 ! ALL EQUATIONS REFER TO THE PAPER BY P. MERKEL (PKM)
38 ! IN J. COMPUT. PHYSICS 66, p83 (1986)
39 !
40 ! IN GOING BETWEEN THE COMPLEX NOTATION OF (PKM) AND OUR REAL FORM,
41 ! NOTE THAT THE INTEGRALS (APPENDIX, PKM) Imn AND Kmn ARE BOTH REAL.
42 ! THUS, THE SIN(mu-nv) INTEGRALS OF THE SINGULAR PIECE (ANALYTIC CONTRIBUTION)
43 ! VANISHES.
44 !
45 ! THE REQUIRED SOURCE-TERM INTEGRALS ARE (Eq.2.16-2.17):
46 !
47 ! BVECS(m,n) = Int< SIN(mu' - nv') han(u',v') >
48 ! BVECC(m,n) = Int< COS(mu' - nv') han(u',v') >
49 !
50 ! Where Int<...> means integration over u (theta) and v (zeta) and
51 ! summation over field periods. These can be written in terms of PKM integrals
52 ! Imn(a,b,c), where a(u,v) = guu (g theta-theta), etc.:
53 !
54 ! BVECS(m,n) = ALP * Int<SIN(mu' - nv') * F * Im,-n(a,b,c)>
55 ! BVECC(m,n) = ALP * Int<COS(mu' - nv') * F * Im,-n(a,b,c)>
56 !
57 ! Here, F = - BNORM(u',v') is defined in Eq.(2.13), and ALP = (2*pi/nfp).
58 !
59 ! Similarly, the analytic part of the matrix A(m,n;m',n') can be written:
60 !
61 ! A(m,n;m',n') = (2*pi/nfp) * Int<SIN(mu' - nv')*SIN(m'u' - n'v')
62 ! [Km,-n](a',b',c';A',B',C')>
63 !
64 ! On EXIT, GRPMN(ip,m,n) = ALP * SIN(ip,m,n) * K[m,-n](ip)
65 !
66 !
67 ! COMPUTE ALL QUANTITIES INDEPENDENT OF THE MODE INDICES L,M,N
68 ! NOTE: 2b = guv_b HAS FACTOR OF 2 BUILT IN (see SUBROUTINE SURFACE)
69 !
70 ! ADP(M): a +(-)2b + c
71 ! CMA: c - a
72 ! DELTA: 4*(ac - b**2)
73 ! AZP(M): A +(-)2*B + C
74 ! CMA1: C - A
75 ! R1P(M): Coefficient of l*Tl+(-) in eq (A17)
76 ! R0P(M): Coefficient of l*T(l-1)+(-) in eq (A17)
77 ! RA1P(M):Coefficient of Tl+(-) in eq (A17)
78 !
79  DO k = nuv3min, nuv3max
80  adp(k) = guu_b(k) + guv_b(k) + gvv_b(k)
81  adm(k) = guu_b(k) - guv_b(k) + gvv_b(k)
82  cma(k) = gvv_b(k) - guu_b(k)
83  sqrtc(k) = two*sqrt(gvv_b(k))
84  sqrta(k) = two*sqrt(guu_b(k))
85  END DO
86 
87  IF (ivacskip .EQ. 0) THEN

89  blksize=blksize_scp*numjs_vac
90  DO l= 1, blksize
91  grpmn(lindx_scp(l)) = 0
92  END DO
93 
94  DO k = nuv3min, nuv3max
95  delt1u(k) = adp(k)*adm(k) - cma(k)*cma(k)
96  azp1u(k) = auu(k) + auv(k) + avv(k)
97  azm1u(k) = auu(k) - auv(k) + avv(k)
98  cma11u(k)= avv(k) - auu(k)
99  r1p(k) = (azp1u(k)*(delt1u(k) - cma(k)*cma(k))/adp(k)
100  1 - azm1u(k)*adp(k) + two*cma11u(k)*cma(k))/delt1u(k)
101  r1m(k) = (azm1u(k)* (delt1u(k) - cma(k)*cma(k))/adm(k)
102  1 - azp1u(k)*adm(k) + two*cma11u(k)*cma(k))/delt1u(k)
103  r0p(k) = (-azp1u(k)*adm(k)*cma(k)/adp(k) - azm1u(k)*cma(k)
104  1 + two*cma11u(k)*adm(k))/delt1u(k)
105  r0m(k) = (-azm1u(k)*adp(k)*cma(k)/adm(k) - azp1u(k)*cma(k)
106  1 + two*cma11u(k)*adp(k))/delt1u(k)
107  ra1p(k) = azp1u(k)/adp(k)
108  ra1m(k) = azm1u(k)/adm(k)
109  END DO
110  ENDIF
111 
112 !
113 ! INITIALIZE VECTORS
114 !
115 ! bvec = 0
116 !
117 ! INITIALIZE T0+ and T0-
118 !
119 ! TLP(M): TL+(-)
120 ! TLP(M)1:T(L-1)+(-)
121 ! TLP(M)2:T(L-2)+(-)
122 !
123  DO k = nuv3min,nuv3max
124  sqad1u(k) = sqrt(adp(k))
125  sqad2u(k) = sqrt(adm(k))
126  tlp1(k) = 0
127  tlm1(k) = 0
128  tlp(k) = one/sqad1u(k)*log((sqad1u(k)*sqrtc(k)
129  1 + adp(k) + cma(k))/(sqad1u(k)*sqrta(k)
130  2 - adp(k) + cma(k)))
131  tlm(k) = one/sqad2u(k)*log((sqad2u(k)*sqrtc(k)
132  1 + adm(k) + cma(k))/(sqad2u(k)*sqrta(k)
133  2 - adm(k) + cma(k)))
134  tlpm(k) = tlp(k) + tlm(k)
135  END DO
136 !
137 ! BEGIN L-SUM IN EQ (A14) TO COMPUTE Imn (and Kmn) INTEGRALS
138 ! NOTE THAT IN THE LOOP OVER L BELOW: L == |m - n| + 2L_A14
139 ! THUS, L BELOW IS THE INDEX OF THE T+- (S+-)
140 !
141  sign1 = 1
142  fl1 = 0
143 
144  lloop: DO l = 0, mf + nf
145  fl = fl1
146 !
147 ! COMPUTE SL+ and SL- , Eq (A17)
148 ! SLP(M): SL+(-)
149 !
150  IF (ivacskip .eq. 0) THEN
151  DO k = nuv3min,nuv3max
152  slp(k) = (r1p(k)*fl + ra1p(k))*tlp(k) + r0p(k)*fl*tlp1(k)
153  1 - (r1p(k) + r0p(k))/sqrtc(k)
154  2 + sign1*(r0p(k) - r1p(k))/sqrta(k)
155  slm(k) = (r1m(k)*fl + ra1m(k))*tlm(k) + r0m(k)*fl*tlm1(k)
156  1 - (r1m(k) + r0m(k))/sqrtc(k)
157  2 + sign1*(r0m(k) - r1m(k))/sqrta(k)
158  slpm(k) = slp(k) + slm(k)
159  END DO
160  ENDIF
161 !
162 ! BEGIN MODE NUMBER (m,n) LOOP
163 !
164  DO n = 0, nf
165  DO m = 0, mf
166 
167  IF (l .EQ. 0) THEN
168  mn = m + mf1*(n+nf) + 1
169  bvec(mn,:) = 0
170  mn = m + mf1*(nf-n) + 1
171  bvec(mn,:) = 0
172  END IF
173 
174  IF (cmns(l,m,n) .eq. zero) cycle
175 
176  IF (n.eq.0 .or. m.eq.0) THEN
177 !
178 ! 1. n = 0 and m >= 0 OR n > 0 and m = 0
179 !
180  CALL analysum_par (grpmn, bvec, slpm, tlpm, m, n, l,
181  1 ivacskip, ndim)
182 
183  ELSE
184 !
185 ! 2. n>=1 and m>=1
186 !
187  CALL analysum2_par (grpmn, bvec, slm, tlm, slp, tlp,
188  1 m, n, l, ivacskip, ndim)
189 
190  ENDIF
191  END DO
192  END DO
193 
194 !
195 ! UPDATE "TL's" (FOR L -> L+1) USING EQ (A15)
196 !
197  fl1 = fl1 + 1 !next l
198  fl2 = 2*fl1-1
199  sign1 = -sign1 !(-1)**l (next l now)
200  DO k = nuv3min, nuv3max
201  tlp2(k) = tlp1(k)
202  tlm2(k) = tlm1(k)
203  tlp1(k) = tlp(k)
204  tlm1(k) = tlm(k)
205  tlp(k) = ((sqrtc(k) + sign1*sqrta(k)) - fl2*
206  1 cma(k)*tlp1(k) - fl*adm(k)*tlp2(k))/(adp(k)*fl1)
207  tlm(k) = ((sqrtc(k) + sign1*sqrta(k)) - fl2*
208  1 cma(k)*tlm1(k) - fl*adp(k)*tlm2(k))/(adm(k)*fl1)
209  tlpm(k) = tlp(k) + tlm(k)
210  END DO
211 
212  END DO lloop
213 
214  DEALLOCATE (r0p, r1p, r0m, r1m, sqrtc, sqrta, tlp2, tlp1,
215  1 tlp, tlm2, tlm1, tlm, adp, adm, cma, ra1p, ra1m, slm,
216  2 slp, tlpm, slpm, delt1u, azp1u, azm1u, cma11u, sqad1u,
217  3 sqad2u, stat = l)
218 
219  CALL second0(skstoff)
220  analyt_time = analyt_time + (skstoff - skston)
221  END SUBROUTINE analyt_par
