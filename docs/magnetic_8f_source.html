399  numrvol = SIZE(rvolgrid, 1)
400  numzvol = SIZE(rvolgrid, 2)
401  ALLOCATE(response_rvol(numrvol, numzvol))
402  ALLOCATE(response_phivol(numrvol, numzvol))
403  ALLOCATE(response_zvol(numrvol, numzvol))
