18  cosm(:,0) = 1

28  END DO
29

35  END IF
36 
37  DO n = 2,ntor
38  cosn(:,n) = cosn(:,n-1)*cosn(:,1)
39  1 - sinn(:,n-1)*sinn(:,1)
40  sinn(:,n) = sinn(:,n-1)*cosn(:,1)
41  1 + cosn(:,n-1)*sinn(:,1)
