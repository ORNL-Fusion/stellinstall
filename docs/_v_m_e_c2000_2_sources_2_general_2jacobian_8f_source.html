119 !
120 ! (RS, ZS)=(R, Z) SUB S, (RU12, ZU12)=(R, Z) SUB THETA(=U)
121 ! AND TAU=SQRT(G)/R ARE DIFFERENCED ON HALF MESH
122 ! NOTE: LOOPS WERE SPLIT TO ALLOW EFFICIENT MEMORY BUS USAGE
123 !
124 ! SQRT(G) = R*TAU IS COMPUTED IN BCOVAR
125 !
126 ! FOR OPTIMIZATION ON CRAY, MUST USE COMPILER DIRECTIVES TO
127 ! GET VECTORIZATION OF LOOPS INVOLVING MORE THAN ONE POINTER!
128 !
129 !
130 ! HERE, TAU = (Ru * Zs - Rs * Zu). THE DERIVATIVES OF SHALF = SQRT(s)
131 ! WERE COMPUTED EXPLICITLY AS: d(shalf)/ds = .5/shalf
132 !
133 ! NOTE: z12 IS USED IN RECONSTRUCTION PART OF CODE ONLY; COULD BE ELIMINATED...
134 !
135 !
136  dphids = p25
137  irst = 1
138 
139 CDIR$ IVDEP
140  DO l = 2,nrzt
141  ru12(l) = p5*(ru(l,meven) + ru(l-1,meven) +
142  1 shalf(l)*(ru(l,modd) + ru(l-1,modd)))
143  zs(l) = ohs*(z1(l,meven) - z1(l-1,meven) +
144  1 shalf(l)*(z1(l,modd) - z1(l-1,modd)))
145  tau(l) = ru12(l)*zs(l) + dphids*
146  1 (ru(l,modd) *z1(l,modd) + ru(l-1,modd) *z1(l-1,modd) +
147  2 (ru(l,meven)*z1(l,modd) + ru(l-1,meven)*z1(l-1,modd))/shalf(l))
148  ENDDO
149 
150 
151 CDIR$ IVDEP
152  DO l = 2,nrzt
153  zu12(l) = p5*(zu(l,meven) + zu(l-1,meven) +
154  1 shalf(l)*(zu(l,modd) + zu(l-1,modd)))
155  rs(l) = ohs*(r1(l,meven) - r1(l-1,meven) +
156  1 shalf(l)*(r1(l,modd) - r1(l-1,modd)))
157  r12(l) = p5*(r1(l,meven) + r1(l-1,meven) +
158  1 shalf(l)*(r1(l,modd) + r1(l-1,modd)))
159  tau(l) = (tau(l) - rs(l)*zu12(l) - dphids*
160  1 (zu(l,modd) *r1(l,modd)+zu(l-1,modd) *r1(l-1,modd)
161  2 + (zu(l,meven)*r1(l,modd)+zu(l-1,meven)*r1(l-1,modd))/shalf(l)))
162  END DO
163

168  tau(1:nrzt:ns) = temp(:)
169  taumax = maxval(tau(2:nrzt))

171  IF (taumax*taumin .lt. zero) irst = 2
172 
173 #if defined(SKS)
174  CALL second0(tjacoff)
175  s_jacobian_time=s_jacobian_time+(tjacoff-tjacon)
176 #endif
177  END SUBROUTINE jacobian
