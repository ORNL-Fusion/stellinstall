351 ! Start of executable code
352 ! Determine the number of integration steps to take by dividing the path length
353 ! by the step length and rounding to the nearest integer.
354  dxcart = vertex2%position - vertex1%position
355  nsteps = int(sqrt(dot_product(dxcart, dxcart))/dxcourse)
356 
357 ! Choose the actual step size.
358  dxcart = dxcart/nsteps
359 
360  search = .false.
361  xcart = vertex1%position
362 
363 ! Linearly search the line until an interval containing the point is detected.
364  DO i = 1, nsteps
365  search = search_function(context, xcart, xcart + dxcart)
366  IF (search) THEN

369  DO WHILE (sqrt(dot_product(dxcart, dxcart)) .gt. dxfine)
370  dxcart = dxcart/2.0
371  IF (.not.search_function(context, xcart, &
372  & xcart + dxcart)) THEN
373  xcart = xcart + dxcart
374  END IF
375  END DO
