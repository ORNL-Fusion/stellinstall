1911  & param_name)

1913  IMPLICIT NONE
1914 
1915 ! Declare Arguments
1916  LOGICAL :: equilibrium_converge
1917  type(equilibrium_class), INTENT(inout) :: this

1919  INTEGER, INTENT(in) :: iou
1920  INTEGER, INTENT(in) :: eq_comm

1922

1925  INTEGER :: error

1928

1935  & error)
1936  CALL mpi_bcast(num_iter, 1, mpi_integer, 0, eq_comm, error)
1937 #endif
1938  IF (this%not_converged .or. this%force_solve) THEN
1939  SELECT CASE (this%type)

1953 
1954  END SELECT

1957  ENDIF
1958

1961 #else
1962  eq_rank = 0
1963 #endif

1965  WRITE (*,1000) equilibrium_converge, num_iter, &
1966  & num_iter - init_num_iter, param_name
1967  WRITE (iou,1000) equilibrium_converge, num_iter, &
1968  & num_iter - init_num_iter, param_name
1969  END IF
1970 
1971  CALL profiler_set_stop_time('equilibrium_converge', start_time)
