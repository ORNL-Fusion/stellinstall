# Create an empty variable to hold all the source files.
set (siesta_sources "")

find_library (SCALAPACK_LIB NAMES scalapack)

# Add subdirectory for all the sources.
add_subdirectory (Sources)

add_library (siesta STATIC ${siesta_sources})
add_dependencies (siesta bmw stell)
set_target_properties (siesta PROPERTIES COMPILE_DEFINITIONS "SKS")
set_target_properties (siesta PROPERTIES Fortran_MODULE_DIRECTORY ${CMAKE_Fortran_MODULE_DIRECTORY}/siesta)
if (CMAKE_VERSION VERSION_GREATER "2.8.8")
	set_target_properties (siesta PROPERTIES INCLUDE_DIRECTORIES "${CMAKE_Fortran_MODULE_DIRECTORY}/stell;${CMAKE_Fortran_MODULE_DIRECTORY}/bmw")
else ()
	set_target_properties (siesta PROPERTIES COMPILE_FLAGS "-I${CMAKE_Fortran_MODULE_DIRECTORY}/stell -I${CMAKE_Fortran_MODULE_DIRECTORY}/bmw")
endif ()

# Define an executable and link all libraries.
add_executable (xsiesta ${CMAKE_CURRENT_SOURCE_DIR}/Sources/siesta.f90)
add_dependencies (xsiesta siesta)
set_target_properties (xsiesta PROPERTIES COMPILE_DEFINITIONS "SKS")
target_link_libraries (xsiesta siesta bmw stell)
set_target_properties (xsiesta PROPERTIES Fortran_MODULE_DIRECTORY ${CMAKE_Fortran_MODULE_DIRECTORY}/xsiesta)

if (MPI_Fortran_FOUND)
        if (CMAKE_VERSION VERSION_GREATER "2.8.8")
                set_target_properties (xsiesta PROPERTIES INCLUDE_DIRECTORIES "${CMAKE_Fortran_MODULE_DIRECTORY}/stell;${CMAKE_Fortran_MODULE_DIRECTORY}/siesta;${MPI_Fortran_INCLUDE_PATH}")
        else ()
                set (mpi_include "")
                FOREACH (path ${MPI_Fortran_INLUCDE_PATH})
                        set (mpi_include "${mpi_include} -I${path}")
                ENDFOREACH (path)
                set_target_properties (xsiesta PROPERTIES COMPILE_FLAGS "-I${CMAKE_Fortran_MODULE_DIRECTORY}/stell ${mpi_inlcude} -I${CMAKE_Fortran_MODULE_DIRECTORY}/siesta")
        endif ()
else ()
        if (CMAKE_VERSION VERSION_GREATER "2.8.8")
                set_target_properties (xsiesta PROPERTIES INCLUDE_DIRECTORIES "${CMAKE_Fortran_MODULE_DIRECTORY}/stell;${CMAKE_Fortran_MODULE_DIRECTORY}/bmw;${CMAKE_Fortran_MODULE_DIRECTORY}/siesta")
        else ()
                set_target_properties (xsiesta PROPERTIES COMPILE_FLAGS "-I${CMAKE_Fortran_MODULE_DIRECTORY}/stell -I${CMAKE_Fortran_MODULE_DIRECTORY}/bmw -I${CMAKE_Fortran_MODULE_DIRECTORY}/siesta")
        endif ()
endif ()

target_link_libraries (xsiesta ${NETCDF_LIBRARIES})
if (SCALAPACK_LIB)
        target_link_libraries (xsiesta ${SCALAPACK_LIB})
endif ()
if (BLAS_FOUND)
        target_link_libraries (xsiesta ${BLAS_LIBRARIES})
endif ()
if (LAPACK_FOUND)
        target_link_libraries (xsiesta ${LAPACK_LIBRARIES})
endif ()
if (MPI_Fortran_FOUND)
        target_link_libraries (xsiesta ${MPI_Fortran_LIBRARIES})
endif ()
