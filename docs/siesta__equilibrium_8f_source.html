735  & SIZE(response_object%a_r)))
736  ALLOCATE(this%magnetic_cache%jzsuv(ns,numu, &
737  & SIZE(response_object%a_r)))
738  END IF

742 ! Set the differental area elements.

747  & / response_object%num_t_shell

752  ALLOCATE(this%magnetic_cache%kphiuv( &
753  & compression_get_dimension1(response_object%a_s_r), &

1938  FUNCTION siesta_get_p_flux(this, flux, normalize)
1939  USE read_wout_mod, only: nfp
1940  USE stel_constants, only: mu0
1941 
1942  IMPLICIT NONE
1943 
1944 ! Declare Arguments
1945  REAL (rprec) :: siesta_get_p_flux
1946  type(siesta_class), INTENT(in) :: this
1947  REAL (rprec), DIMENSION(3), INTENT(in) :: flux
1948  LOGICAL, INTENT(in) :: normalize
1949

1953  REAL (rprec) :: whigh

1961

1963  start_time = profiler_get_start_time()
1964 
1965  ds = 1.0/(this%context%ns - 1.0)
1966 
1967  IF (flux(1) .lt. ds/2.0) THEN
1968  ilow = 2
1969  ihigh = 3
1970  whigh = flux(1) - ds/2.0
1971  wlow = 1.0 - whigh
1972  ELSE IF (flux(1) .gt. 1.0 - ds/2.0) THEN
1973  ihigh = this%context%ns
1974  ilow = ihigh - 1
1975  wlow = -flux(1)/ds - 0.5 + 1.0/ds
1976  whigh = 1.0 - wlow
1977  ELSE
1978  ilow = floor(flux(1)/ds + 1.5)
1979  ihigh = ilow + 1
1980  whigh = flux(1)/ds + 1.5 - ilow
1981  wlow = 1.0 - whigh
1982  END IF
1983 
1984  mpol = this%context%mpol
1985  ntor = this%context%ntor
1986 
1987  siesta_get_p_flux = 0.0
1988  DO n = -ntor, ntor
1989  DO m = 0, mpol
1990  siesta_get_p_flux = siesta_get_p_flux + &
1991  & (wlow*this%context%pmnch(m,n,ilow) + &
1992  & whigh*this%context%pmnch(m,n,ihigh))* &
1993  & cos(m*flux(2) + nfp*n*flux(3))
1994

2010 
2011  siesta_get_p_flux = (siesta_get_p_flux - this%context%p_min) &
2012  & / (this%context%p_max - this%context%p_min)
2013  ELSE
2014  siesta_get_p_flux = siesta_get_p_flux &
2015  & / (mu0*this%context%p_factor)
2016  END IF
2017 
2018  CALL profiler_set_stop_time('siesta_get_p_flux', start_time)
