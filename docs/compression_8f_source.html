94  FUNCTION compression_construct_new(data_in, svd_cut_off)
95  USE v3_utilities
96 
97  IMPLICIT NONE
98 
99 ! Declare Arguments
100  type(compression_class), POINTER :: compression_construct_new
101  REAL (rprec), DIMENSION(:,:), INTENT(in) :: data_in
102  REAL (rprec), INTENT(in) :: svd_cut_off
103 
104 ! local arguments
105  INTEGER :: m, n, i
106  REAL (rprec), DIMENSION(:), ALLOCATABLE :: w_svd
107  REAL (rprec), DIMENSION(:), ALLOCATABLE :: svd_work
108  INTEGER :: work_size
109  INTEGER :: status
110  REAL (rprec) :: e_zero
111  REAL (rprec) :: e_r
112 
113 ! Start of executable code
114  ALLOCATE(compression_construct_new)
115 
116 ! Special case if all the data is zero, there is no need to retain any singular
117 ! values. This will be detected in the uncompress by having the data_dim array
118 ! allocated.
119  IF (all(data_in .eq. 0.0)) THEN
120  ALLOCATE(compression_construct_new%data_dim(2))
121  compression_construct_new%data_dim = shape(data_in)
122  RETURN
123  END IF
124 
125  m = SIZE(data_in, 1)
126  n = SIZE(data_in, 2)
127  ALLOCATE(compression_construct_new%data_buffer(m,n))
128  compression_construct_new%data_buffer = data_in
129 
130  IF (svd_cut_off .eq. 0.0) THEN
131  RETURN
132  END iF
133 
134  ALLOCATE(compression_construct_new%wvt_buffer(n,n))
135  ALLOCATE(compression_construct_new%u_buffer(m,m))
136  ALLOCATE(svd_work(1))
137  ALLOCATE(w_svd(min(m,n)))
138 
139  compression_construct_new%wvt_buffer = 0.0
140  compression_construct_new%u_buffer = 0.0
141  w_svd = 0.0
142  svd_work = 0.0
143 
144 ! Find the optimal work size.
145  CALL dgesvd('All', 'All', m, n, &
146  & compression_construct_new%data_buffer, m, &
147  & w_svd, compression_construct_new%u_buffer, m, &
148  & compression_construct_new%wvt_buffer, n, svd_work, -1, &
149  & status)
150  work_size = int(svd_work(1))
151  DEALLOCATE(svd_work)
152  ALLOCATE(svd_work(work_size))
153  svd_work = 0.0
154 
155 ! Factor the matrix to M = U * W * V^T
156  CALL dgesvd('All', 'All', m, n, &
157  & compression_construct_new%data_buffer, m, &
158  & w_svd, compression_construct_new%u_buffer, m, &
159  & compression_construct_new%wvt_buffer, n, svd_work, &
160  & work_size, status)
161  CALL assert_eq(0, status, 'dgesvd problem when compressing ' // &
162  & 'buffer')
163 
164  DEALLOCATE(svd_work)
165 
166 ! Determine the number of singular values to keep. If the number of singular
167 ! values is greater than half, do not compress the buffer. First start by
168 ! setting any negative values to zero.
169  WHERE (w_svd .lt. 0.0)
170  w_svd = 0.0
171  END WHERE
172 
173 ! Find the minimum number of singular values to keep based on
174 ! 1 - n(r) = svd_cut_off where n(r) is equation (7) in del-Castillo-Negrete
175 ! et. al. doi:10.1016/j.jcp.2006.07.022
176  e_zero = dot_product(w_svd, w_svd)
177 
178  e_r = 0.0
179 ! We only need to check up to one more than half the number of singular values.
180 ! Anymore than that will use more data storage. Store the number of singular
181 ! values retained in the work_size.
182  DO i = 1, SIZE(w_svd)/2 + 1
183  e_r = e_r + w_svd(i)*w_svd(i)
184  work_size = i
185  IF ((1.0 - e_r/e_zero) .le. svd_cut_off) THEN
186  EXIT
187  END IF
188  END DO
189 
190 ! Check the compression percentage. The compression is defined as the
191 ! percentage of the orginal data size. The compression percentage is defined as
192 !
193 ! compressed_size/uncompressed_size*100.0 (1)
194 !
195 ! where
196 !
197 ! compressed_size = r + r*m + r*n (2)
198 !
199 ! and
200 !
201 ! uncompressed_size = n*m (3)
202 !
203 ! n and m are the dimensions of the orginal matrix and r is the number of
204 ! singular values retained.
205 !
206 ! However you can gain even more storage space. The compressed data is '
207 ! uncompressed reconstructing the matrix from U.W.V^T. By precomputing W.V^T
208 ! and storing that buffer instead instead of the W and V^T buffers separately,
209 ! the an array the size of the number of signular values is eliminiated. This
210 ! also has the added benifit of speed when uncompressing the data. The new
211 ! compressed size becomes
212 !
213 ! compressed_size = r*m + r*n (4)
214 !
215 ! Barrow the e_ variables to compute th compression
216 ! percentage. Use e_r for the compressed size and e_zero of the uncompressed
217 ! size.
218 
219  e_r = work_size*m + work_size*n
220  e_zero = n*m
221 
222  IF (100.0*e_r/e_zero .ge. compression_max_percentage) THEN
223 ! Use the uncompressed buffers.
224  compression_construct_new%data_buffer = data_in
225 
226  DEALLOCATE(compression_construct_new%u_buffer)
227  compression_construct_new%u_buffer => null()
228 
229  DEALLOCATE(compression_construct_new%wvt_buffer)
230  compression_construct_new%wvt_buffer => null()
231  ELSE
232 ! Use the compressed buffers. Barrow the data_buffer as temp buffer for
233 ! resizing the storage buffers.
234  DEALLOCATE(compression_construct_new%data_buffer)
235 
236 ! Store the U buffer. The reduced U buffer now only needs m x r storage space.
237  ALLOCATE(compression_construct_new%data_buffer(m,work_size))
238  compression_construct_new%data_buffer = &
239  & compression_construct_new%u_buffer(:,1:work_size)
240  DEALLOCATE(compression_construct_new%u_buffer)
241  ALLOCATE(compression_construct_new%u_buffer(m,work_size))
242  compression_construct_new%u_buffer = &
243  & compression_construct_new%data_buffer
244  DEALLOCATE(compression_construct_new%data_buffer)
245 
246 ! Store the VT buffer. The reduced VT buffer now only needs r x n storage
247 ! space.
248  ALLOCATE(compression_construct_new%data_buffer(work_size,n))
249  compression_construct_new%data_buffer = &
250  & compression_construct_new%wvt_buffer(1:work_size,:)
251  DEALLOCATE(compression_construct_new%wvt_buffer)
252  ALLOCATE(compression_construct_new%wvt_buffer(work_size,n))
253 
254 ! Multiply W.VT and store in the WVT buffer.
255  DO i = 1, n
256  compression_construct_new%wvt_buffer(:,i) = &
257  & w_svd(1:work_size) * &
258  & compression_construct_new%data_buffer(:,i)
259  END DO
260 
261  DEALLOCATE(compression_construct_new%data_buffer)
262  compression_construct_new%data_buffer => null()
263  END IF
264 
265  DEALLOCATE(w_svd)
266 
267  END FUNCTION
