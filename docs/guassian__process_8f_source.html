353  SUBROUTINE gaussp_set_profile(this, a_model)
354 
355  IMPLICIT NONE
356 
357 ! Declare Arguments
358  type(gaussp_class), INTENT(inout) :: this
359  type(model_class), INTENT(inout) :: a_model
360 
361 ! Local Variables
362  INTEGER :: i
363  INTEGER :: j
364  REAL (rprec) :: start_time
365  INTEGER :: ierr
366  REAL (rprec), DIMENSION(:), ALLOCATABLE :: cached_value
367  REAL (rprec), DIMENSION(:), ALLOCATABLE :: gradient
368  REAL (rprec) :: evidence
369  REAL (rprec) :: new_evidence
370 
371 ! local parameters
372  REAL(rprec), PARAMETER :: gamma = 1.0
373 
374 ! Start of executable code
375  start_time = profiler_get_start_time()
376 
377 ! Flags will have a single bit position of a corresponding profile set.
378 !
379 ! this%flags .and. a_model%state_flags
380 !
381 ! will value to true if the coresponding flag is set in the model state flags.
382  IF (btest(a_model%state_flags, model_state_vmec_flag) .or. &
383  & btest(a_model%state_flags, model_state_siesta_flag) .or. &
384  & btest(a_model%state_flags, model_state_shift_flag) .or. &
385  & btest(a_model%state_flags, model_state_signal_flag) .or. &
386  & iand(this%flags, a_model%state_flags) .ne. 0) THEN
387 
388 ! Use gradient ascent to fine the optimal hyper parameters by maximizing the
389 ! the evidence.
390  new_evidence = gaussp_get_evidence(this, a_model)
391 
392  ALLOCATE(cached_value(SIZE(this%params)))
393  ALLOCATE(gradient(SIZE(this%params)))
394 
395  DO i = 1, SIZE(this%params)
396  cached_value(i) = param_get_value(this%params(i)%p, a_model)
397  END DO
398 
399  WRITE (this%iou,1000)
400  WRITE (this%iou,1001) new_evidence
401 
402  DO
403  evidence = new_evidence
404 
405 ! Compute gradient.
406  DO i = 1, SIZE(this%params)
407  CALL param_increment(this%params(i)%p, a_model, &
408  & mpi_comm_null, .false.)
409  gradient(i) = gaussp_get_evidence(this, a_model) &
410  & - evidence
411 
412  CALL param_increment(this%params(i)%p, a_model, &
413  & mpi_comm_null, .false.)
414  END DO
415 
416 ! Step towards the maximum.
417  cached_value = cached_value + gamma*gradient
418  DO i = 1, SIZE(this%params)
419  CALL param_set_value(this%params(i)%p, a_model, &
420  & cached_value(i), mpi_comm_null, &
421  & .false.)
422  END DO
423  new_evidence = gaussp_get_evidence(this, a_model)
424 
425  WRITE (this%iou,1002) new_evidence, new_evidence - evidence
426  FLUSH(this%iou)
427 
428  IF (abs(new_evidence - evidence) < this%tolerance) EXIT
429  END DO
430  END IF
431 
432  WRITE (this%iou,*) 'Final Parameters', cached_value
433  FLUSH(this%iou)
434 
435  DEALLOCATE(cached_value)
436  DEALLOCATE(gradient)
437 
438 ! Found the maximum now set the profile. The profile is determined by sampling
439 ! the distribution.
440 !
441 ! af = K_SL * A^-1 (1)
442 !$OMP PARALLEL
443 !$OMP& DEFAULT(SHARED)
444 !$OMP DO
445 !$OMP& SCHEDULE(STATIC)
446  DO i = 1, SIZE(this%fpoints)
447  this%fpoints(i) = dot_product(this%kls(:,i), this%work(:,1))
448  END DO
449 !$OMP END DO
450 !$OMP END PARALLEL
451 
452  CALL profiler_set_stop_time('gaussp_get_modeled_signal', &
453  & start_time)
454 
455 1000 FORMAT('New Process')
456 1001 FORMAT('Log Evidence : ',es12.5)
457 1002 FORMAT('Log Evidence : ',es12.5,' Change :',es12.5)

474  FUNCTION gaussp_get_evidence(this, a_model)
475  USE stel_constants, ONLY: twopi
476 
477  IMPLICIT NONE
478 
479 ! Declare Arguments
480  REAL (rprec) :: gaussp_get_evidence
481  type(gaussp_class), INTENT(inout) :: this
482  type(model_class), INTENT(inout) :: a_model
483 
484 ! Local Variables
485  INTEGER :: i
486  INTEGER :: j
487  REAL (rprec) :: start_time
488  INTEGER :: ierr
489 
490 ! Local Parameters
491  REAL (rprec), PARAMETER :: log2pi = log(twopi)
492 
493 ! Start of executable code
494  start_time = profiler_get_start_time()
495 
496 !$OMP PARALLEL
497 !$OMP& DEFAULT(SHARED)
498 !$OMP& PRIVATE(i,j)
499 
500 !$OMP DO
501 !$OMP& SCHEDULE(DYNAMIC)
502  DO j = 1, SIZE(this%fpoints)
503  DO i = 1, SIZE(this%signals)
504  this%kls(i,j) = signal_get_gp(this%signals(i)%p, a_model, j, &
505  & this%flags)
506  this%work(i,j + 1) = this%kls(i,j)
507  END DO
508  END DO
509 !$OMP END DO
510 
511 !$OMP DO
512 !$OMP& SCHEDULE(DYNAMIC)
513  DO j = 1, SIZE(this%signals)
514  DO i = j, SIZE(this%signals)
515  this%kll(i,j) = signal_get_gp(this%signals(i)%p, a_model, &
516  & this%signals(j)%p, this%flags)
517  IF (j .ne. i) THEN
518  this%kll(j,i) = this%kll(i,j)
519  END IF
520  END DO
521  this%kll(j,j) = this%kll(j,j) &
522  & + signal_get_sigma2(this%signals(j)%p) &
523  & + this%cholesky_fac
524  this%work(j,1) = signal_get_observed_signal(this%signals(j)%p, &
525  & a_model)
526  END DO
527 !$OMP END DO
528 !$OMP END PARALLEL
529 
530 ! CALL LAPACK to do Cholesky factorization of the A=(K_LL + Sigma_y)
531  CALL dpotrf('L', SIZE(this%signals), this%kll, &
532  & SIZE(this%signals), ierr)
533 
534 ! The elements in the cholesky matrix for j > i are meaning less.
535  IF (ierr .lt. 0) THEN
536  CALL err_fatal('gaussp_get_modeled_signal: DROTRF ' // &
537  & 'cannot factor matrix. ierr is negative')
538  ELSE IF (ierr .gt. 0) THEN
539  CALL err_fatal('gaussp_get_modeled_signal: DROTRF ' // &
540  & 'cannot factor matrix. ierr is ' // &
541  & 'positive. Try increasing gp_cholesky_fact')
542  END IF
543 
544 ! IF statement correctly throws an error if I cannot factor the matrix!
545 
546 ! CALL LAPACK to solve A^-1y and A^-1 K_LS simultaneously. The work matrix
547 ! stores the RHS on input, and returns the solution matrix.
548  CALL dpotrs('L', SIZE(this%signals), SIZE(this%fpoints) + 1, &
549  & this%kll, SIZE(this%signals), this%work, &
550  & SIZE(this%signals), ierr)
551  IF (ierr .lt. 0) THEN
552  CALL err_fatal('gaussp_get_modeled_signal: DROTRS ' // &
553  & 'failed to solve the equation')
554  END IF
555 
556 ! Calculate log evidence.
557 !
558 ! P(y|sigma,x) = 1/Sqrt((2Pi)^N*|A|)Exp(-y^T.A^-1.y/2) (1)
559 !
560 ! Taking the ln of the evidence results in.
561 !
562 ! ln(P) = -N/2*ln(2Pi) - ln(|A|)/2 - y^T.A^-1.y/2 (2)
563 !
564 ! The determinant of A is found by the Cholesky factorization and equals the
565 ! product of the squares of the diagonal.
566 !
567 ! |A| = Prod (A_ii)^2 (3)
568 !
569 ! Taking the ln of this results in.
570 !
571 ! ln(|A|) = Sum 2*ln(A_ii) (4)
572 !
573 
574  gaussp_get_evidence = -0.5*SIZE(this%signals)*log2pi
575 !$OMP PARALLEL
576 !$OMP& DEFAULT(SHARED)
577 !$OMP& PRIVATE(j)
578 !$OMP DO
579 !$OMP& REDUCTION(-:gaussp_get_evidence)
580  DO j = 1, SIZE(this%signals)
581  gaussp_get_evidence = gaussp_get_evidence &
582  & - log(this%kll(j,j)) &
583  & - 0.5*signal_get_observed_signal( &
584  & this%signals(j)%p, a_model)* &
585  & this%work(j,1)
586  END DO
587 !$OMP END DO
588 !$OMP END PARALLEL
589 
590  CALL profiler_set_stop_time('gaussp_get_evidence', start_time)
591 
592  END FUNCTION
