1  SUBROUTINE fouri (grpmn, gsource, amatrix, amatsq, bvec,
2  1 bvecNS, ndim)
3  USE vacmod
4  USE parallel_include_module
5  USE timer_sub
6  IMPLICIT NONE
7 C-----------------------------------------------
8 C D u m m y A r g u m e n t s
9 C-----------------------------------------------
10  INTEGER, INTENT(IN) :: ndim
11  REAL(dp), DIMENSION(mnpd2,nuv3), INTENT(IN) :: grpmn
12  REAL(dp), DIMENSION(nuv), INTENT(IN) :: gsource
13  REAL(dp), DIMENSION(mnpd,mnpd,ndim**2), INTENT(OUT) :: amatrix
14  REAL(dp), DIMENSION(mnpd2,mnpd2), INTENT(OUT) :: amatsq
15  REAL(dp), DIMENSION(mnpd,ndim), INTENT(INOUT) :: bvec, bvecns
16 C-----------------------------------------------
17 C L o c a l P a r a m e t e r s
18 C-----------------------------------------------
19 C interior (int_ext=-1), exterior (int_ext=+1) neumann problem
20  REAL(dp), PARAMETER :: int_ext = 1
21 C-----------------------------------------------
22 C L o c a l V a r i a b l e s
23 C-----------------------------------------------
24  INTEGER :: i, j, k, m, mn, mn0, n
25  REAL(dp), ALLOCATABLE, DIMENSION(:) :: source
26  REAL(dp) :: ton, toff, tfourion, tfourioff
27 !-----------------------------------------------
28 !
29 ! AMATRIX(,1) = A(Sin)(Sin'); AMATRIX(,2) = A(Sin)(Cos');
30 ! AMATRIX(,3) = A(Cos)(Sin'); AMATRIX(,4) = A(Cos)(Cos')
31 !
32 ! ARG OF TRIG FUNCTIONS: m'u' - n'v' CORRESPONDING TO THE PRIMED MESH IN
33 ! PKM EQ.(2.14), AND mu - nv (UNPRIMED MESH) IN EQ. (2.16)
34 !
35 ! ON ENTRY, GRPMN(MN,...) HAS BEEN FOURIER-TRANSFORMED OVER THE UNPRIMED
36 ! COORDINATES (IN FOURP), SO THE FIRST INDEX OF GRPMN CORRESPONDS TO THE FIRST
37 ! INDEX OF AMATRIX. THUS, THE FOURIER TRANSFORMS OF GRPMN HERE ARE OVER THE PRIMED MESH.
38 !
39 ! IN CONTRAST, THE INTEGRAL OF THE SOURCE TERM OVER THE PRIMED MESH WAS ALREADY
40 ! DONE (IN SCALPOT), SO HERE THE FT ARE OVER THE UNPRIMED MESH FOR THE SOURCE.
41 !
42  CALL second0(tfourion)
43 
44  ALLOCATE (source(nuv3), stat=i)
45  IF (i .NE. 0) stop 'Allocation error in fouri'
46 !
47 ! STELLARATOR-SYMMETRIZE SOURCE TERMS (with respect to u,v and -u,-v)
48 ! INDEX (1) IS ANTI-SYMMETRIC, INDEX (2) IS SYMMETRIC
49 !
50 ! GSOURCE = - (2pi)**2 B * dS (h - hsing) * WINT
51 !
52 ! WINT: needed to normalize integral over angles to unity
53 !
54  IF (lasym) THEN
55  source(nuv3min:nuv3max) = onp*gsource(nuv3min:nuv3max)
56  ELSE
57  k = 0
58  DO i = 1, nu2
59  DO j = 1, nv
60  k = k + 1
61  IF (nuv3min.LE.k .AND. k.LE.nuv3max) THEN
62  source(k) = p5*onp*(gsource(k) - gsource(imirr(k)))
63  END IF
64  END DO
65  END DO
66  END IF
67 
68 !
69 ! INITIALIZE RUNNING-SUM ARRAYS
70 !
71  bvecns = 0
72  amatrix = 0
73 
74 !
75 ! PERFORM M,N TRANSFORMS
76 !
77  mn = 0
78  nloop2: DO n = -nf, nf
79  mloop: DO m = 0, mf
80  mn = mn+1
81  j = 0
82  IF (m.EQ.0 .AND. n.LT.0) cycle
83  DO i = nuv3min, nuv3max
84  j = j+1
85  bvecns(mn,1) = bvecns(mn,1)
86  1 + sinmni(j,mn)*source(i)
87 
88  amatrix(:,mn,1) = amatrix(:,mn,1)
89  1 + sinmni(j,mn)*grpmn(:mnpd,i)
90 
91  IF (.NOT.lasym) cycle
92 
93  bvecns(mn,2) = bvecns(mn,2)
94  1 + cosmni(j,mn)*source(i)
95 
96  amatrix(:,mn,2) = amatrix(:,mn,2)
97  1 + cosmni(j,mn)*grpmn(:mnpd,i)
98  amatrix(:,mn,3) = amatrix(:,mn,3)
99  1 + sinmni(j,mn)*grpmn(mnpd+1:,i)
100  amatrix(:,mn,4) = amatrix(:,mn,4)
101  1 + cosmni(j,mn)*grpmn(mnpd+1:,i)
102  END DO
103  END DO mloop
104  END DO nloop2
105 
106 #if defined(SKS)
107  IF (vlactive) THEN
108  CALL second0(ton)
109  CALL mpi_allreduce(mpi_in_place,amatrix,SIZE(amatrix),mpi_real8,
110  1 mpi_sum,vac_comm,mpi_err)
111  CALL second0(toff)
112  allreduce_time = allreduce_time + (toff - ton)
113  END IF
114 #endif
115 !
116 ! ADD (still not reduced) ANALYTIC AND NON-SINGULAR PARTS
117 !
118  bvec = bvec + bvecns
119 
120  DEALLOCATE (source, stat=i)
121 
122  mn0 = 1+mf1*nf !Index of m,n=(0,0)
123 
124 !SANITY CHECKS
125 ! IF (ANY(bvec(1:mn0-mf1:mf1,:) .NE. 0._dp)) STOP 'BVEC != 0'
126 ! IF (ANY(amatrix(:,1:mn0-mf1:mf1,:) .ne. 0._dp)) STOP 'AMAT1 != 0'
127 ! IF (ANY(amatrix(1:mn0-mf1:mf1,:,:) .ne. 0._dp)) STOP 'AMAT2 != 0'
128 !
129 ! ADD DIAGONAL TERMS TO AMATRIX [THE FIRST TERM IN EQ(3.2) OF PKM]
130 !
131  DO mn = 1, mnpd
132  amatrix(mn,mn,1) = amatrix(mn,mn,1) + pi3*int_ext
133  END DO
134 
135  IF (lasym) THEN
136  DO mn = 1, mnpd
137  amatrix(mn,mn,4) = amatrix(mn,mn,4) + pi3*int_ext
138  END DO
139  amatrix(mn0,mn0,4) = amatrix(mn0,mn0,4) + pi3*int_ext !!COS(0-0) mode *2
140  END IF
141 
142 !
143 ! PUT ELEMENTS INTO SQUARE MATRIX
144 !
145  amatsq(:mnpd,:mnpd) = amatrix(:,:,1) !Sin-Sin'
146 
147  IF (lasym) THEN
148 
149  amatsq(:mnpd,1+mnpd:mnpd2) = amatrix(:,:,2) !Sin-Cos'
150  amatsq(1+mnpd:mnpd2,:mnpd) = amatrix(:,:,3) !Cos-Sin'
151  amatsq(1+mnpd:mnpd2,1+mnpd:mnpd2) = amatrix(:,:,4) !Cos-Cos'
152 
153  END IF
154 
155  CALL second0(tfourioff)
156  timer_vac(tfouri) = timer_vac(tfouri) + (tfourioff-tfourion)
157  fouri_time = timer_vac(tfouri)
158 
159  END SUBROUTINE fouri
