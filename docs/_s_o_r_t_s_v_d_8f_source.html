2 c------------------------------
3 c Sorts the weights and U, V matrices from SVD decomposition
4 c Sorts weights in DECREASING order
5 c Based on routine SORT2 from Numerical Recipes, pg 231
6 c Created by Prashant Valanju, (Sept 1998)
7 c------------------------------
8  USE stel_kinds
9  IMPLICIT NONE
10 C-----------------------------------------------
11 C D u m m y A r g u m e n t s
12 C-----------------------------------------------
13  INTEGER m, n, mp, np
14  REAL(rprec), DIMENSION(NP) :: w
15  REAL(rprec), DIMENSION(MP,NP) :: u
16  REAL(rprec), DIMENSION(NP,NP) :: v
17 C-----------------------------------------------
18 C L o c a l V a r i a b l e s
19 C-----------------------------------------------
20  INTEGER :: l, ir, i, j
21  REAL(rprec), DIMENSION(MP) :: ru
22  REAL(rprec), DIMENSION(NP) :: rv
23  REAL(rprec) :: rw
24 C-----------------------------------------------
25  IF (m .GT. mp) stop 'M > MP IN SORTSVD'
26  IF (n .GT. np) stop 'N > NP IN SORTSVD'
27 
28  l = n/2 + 1
29  ir = n
30  10 CONTINUE
31  IF (l .gt. 1) THEN
32  l = l - 1
33  rw = w(l)
34  ru(:m) = u(:m,l)
35  rv(:n) = v(:n,l)
36  ELSE
37  rw = w(ir)
38  ru(:m) = u(:m,ir)
39  rv(:n) = v(:n,ir)
40  w(ir) = w(1)
41  u(:m,ir) = u(:m,1)
42  v(:n,ir) = v(:n,1)
43  ir = ir - 1
44  IF (ir .eq. 1) THEN
45  w(1) = rw
46  u(:m,1) = ru(:m)
47  v(:n,1) = rv(:n)
48  RETURN
49  ENDIF
50  ENDIF
51  i = l
52  j = l + l
53  20 CONTINUE
54  IF (j .le. ir) THEN
55  IF (j .lt. ir) THEN
56 C !Just change these 2 for increasing
57  IF (w(j) .gt. w(j+1)) j = j + 1
58  ENDIF
59  IF (rw .gt. w(j)) THEN !Just change these 2 for increasing
60  w(i) = w(j)
61  u(:m,i) = u(:m,j)
62  v(:n,i) = v(:n,j)
63  i = j
64  j = j + j
65  ELSE
66  j = ir + 1
67  ENDIF
68  GO TO 20
69  ENDIF
70  w(i) = rw
71  u(:m,i) = ru(:m)
72  v(:n,i) = rv(:n)
73  GO TO 10
74 
75  END SUBROUTINE sortsvd
