34  SUBROUTINE update_bfield (l_add_res)
35 !
36 ! PURPOSE: ADVANCES HALF-MESH MAGNETIC FIELD COMPONENTS IN MN SPACE FROM TIME T TO T+DELTA_T
37 ! (Used to compute dBSUPXMNH for updating BSUPXMNH)
38 ! DISCRETIZED VERSION OF FARADAY'S LAW dB/dt = -curl(E)
39 !
40  USE quantities
41  USE descriptor_mod, ONLY: inhessian
42  USE siesta_namelist, ONLY: lresistive, eta_factor
43  USE shared_data, ONLY: lasym, fsq_res, fsq_total, buv_res
44 !-----------------------------------------------
45 ! D u m m y A r g u m e n t s
46 !-----------------------------------------------
47  LOGICAL, INTENT(IN) :: l_add_res
48 !-----------------------------------------------
49 ! L o c a l V a r i a b l e s
50 !-----------------------------------------------
51  INTEGER :: istat, js
52  LOGICAL :: ladd_pert, ltest=.true.
53  REAL(dp), DIMENSION(:,:,:), ALLOCATABLE :: resistivity
54  REAL(dp) :: ton, toff, rho
55  REAL(dp) :: delt_cfl, eta_prof
56  !-----------------------------------------------
57  CALL second0(ton)
58 
59  nsmin=max(1,startglobrow-1); nsmax=min(ns,endglobrow+1)
60 
61 !
62 !Calculate covariant components of (non-resistive) electric field E = -v X B on full-space mesh
63 !
64  ALLOCATE(esubsijf(ntheta,nzeta,nsmin:nsmax), &
65  esubuijf(ntheta,nzeta,nsmin:nsmax), &
66  esubvijf(ntheta,nzeta,nsmin:nsmax), stat=istat)
67  CALL assert(istat.EQ.0,'Allocation1 failed in UPDATE_BFIELD')
68 
69  ALLOCATE(de_uds(0:mpol,-ntor:ntor,nsmin:nsmax), &
70  de_vds(0:mpol,-ntor:ntor,nsmin:nsmax), stat=istat)
71  CALL assert(istat.EQ.0,'Allocation2 failed in UPDATE_BFIELD')
72 
73  esubsijf =-(jvsupuijf(:,:,nsmin:nsmax)*bsupvijf0(:,:,nsmin:nsmax) &
74  - jvsupvijf(:,:,nsmin:nsmax)*bsupuijf0(:,:,nsmin:nsmax)) ! -E_s = V^uB^v - V^vB^u (SINE): jv -> jac*v
75  esubuijf =-(jvsupvijf(:,:,nsmin:nsmax)*bsupsijf0(:,:,nsmin:nsmax) &
76  - jvsupsijf(:,:,nsmin:nsmax)*bsupvijf0(:,:,nsmin:nsmax)) ! -E_u = V^vB^s - V^sB^v (COSINE)
77  esubvijf =-(jvsupsijf(:,:,nsmin:nsmax)*bsupuijf0(:,:,nsmin:nsmax) &
78  - jvsupuijf(:,:,nsmin:nsmax)*bsupsijf0(:,:,nsmin:nsmax)) ! -E_v = V^sB^u - V^uB^s (COSINE)
79 
80 !verify boundary condition: esubu,v(s=1) = 0 (tangential E vanishes at bdy => dB^s = 0)
81  IF (nsmax .EQ. ns) ltest = all(esubuijf(:,:,ns).EQ.zero)
82  CALL assert(ltest,'esubuijf(ns) != 0 in UPDATE_BFIELD')
83  IF (nsmax .EQ. ns) ltest = all(esubvijf(:,:,ns).EQ.zero)
84  CALL assert(ltest,'esubvijf(ns) != 0 in UPDATE_BFIELD')
85 
86 !
87 !Note this will LOWER the energy due to eta*|J|||**2 heating
88 ! esubX(resistive) = eta(JdotB/B^2)*BsubX
89 !so the magnetic energy DECREASES due to this term. Note ksubX=JsubX are the
90 !covariant components of the current
91 !
92  ladd_pert = ALLOCATED(buv_res)
93  IF (lresistive .AND. (l_add_res .or. ladd_pert)) THEN
94  delt_cfl = hs_i**2*abs(eta_factor)
95  IF (fsq_total < fsq_res) delt_cfl = delt_cfl*sqrt(fsq_total/fsq_res)
96 
97  ALLOCATE(resistivity(ntheta,nzeta,nsmin:nsmax), stat=istat)
98  CALL assert(istat.EQ.0,'Allocation3 failed in UPDATE_BFIELD')
99 
100  DO js = nsmin, nsmax
101  rho = hs_i*(js-1)
102  eta_prof = rho*rho*(1-rho)
103  resistivity(:,:,js) = delt_cfl*eta_prof
104  END DO
105 
106  IF (ladd_pert) THEN
107  resistivity = resistivity*buv_res(:,:,nsmin:nsmax)
108  ELSE
109  resistivity = resistivity/jacobf(:,:,nsmin:nsmax) !divide out jacobf factor from cv_currents
110  ENDIF
111 
112 !Isotropic resistivity, E ~ eta*J; for ladd_pert=TRUE, K ~ B in init_state
113  esubsijf = esubsijf + resistivity*ksubsijf(:,:,nsmin:nsmax)
114  esubuijf = esubuijf + resistivity*ksubuijf(:,:,nsmin:nsmax)
115  esubvijf = esubvijf + resistivity*ksubvijf(:,:,nsmin:nsmax)
116 
117  DEALLOCATE (resistivity)
118  ENDIF
119 
120 !Allocate working arrays used in Faraday
121  ALLOCATE(esubsmnf(0:mpol,-ntor:ntor,nsmin:nsmax), &
122  esubumnf(0:mpol,-ntor:ntor,nsmin:nsmax), &
123  esubvmnf(0:mpol,-ntor:ntor,nsmin:nsmax), &
124  esubsmnh(0:mpol,-ntor:ntor,nsmin:nsmax), &
125  esubumnh(0:mpol,-ntor:ntor,nsmin:nsmax), &
126  esubvmnh(0:mpol,-ntor:ntor,nsmin:nsmax), stat=istat)
127  CALL assert(istat.EQ.0,'Allocation4 failed in UPDATE_BFIELD')
128 
129 !Update Bfield using Faraday's Law
130  CALL faraday(djbsupsmnsh, djbsupumnch, djbsupvmnch, isym)
131  IF (lasym) &
132  CALL faraday(djbsupsmnch, djbsupumnsh, djbsupvmnsh, iasym)
133 
134  DEALLOCATE (esubsijf, esubuijf, esubvijf, de_uds, de_vds, &
135  esubsmnf, esubumnf, esubvmnf, &
136  esubsmnh, esubumnh, esubvmnh, stat=istat)
137 
138  CALL second0(toff)
139  time_update_bfield = time_update_bfield + (toff-ton)
140 
141  END SUBROUTINE update_bfield

147  SUBROUTINE faraday(djbsupsmnh, djbsupumnh, djbsupvmnh, ipar)
148  USE shared_data, ONLY: l_natural, delta_t
149  USE hessian, ONLY: l_compute_hessian
150 !-----------------------------------------------
151 ! D u m m y A r g u m e n t s
152 !-----------------------------------------------
153  INTEGER, INTENT(IN) :: ipar
154  REAL(dp),DIMENSION(:,:,:), ALLOCATABLE :: &
155  djbsupsmnh, djbsupumnh, djbsupvmnh
156 !-----------------------------------------------
157 ! L o c a l V a r i a b l e s
158 !-----------------------------------------------
159  INTEGER :: fours, fouru, fourv, m, n, nmin, nmax
160  REAL(dp) :: sparity, r12, mp, np
161 !-----------------------------------------------
162  nmin=nsmin; nmax=nsmax
163 
164  IF (ipar .EQ. isym) THEN !e_s (sin), e_u (cos), e_v (cos)
165  fours=f_sin; fouru=f_cos; fourv=f_cos; sparity = 1
166  ELSE !e_s (cos), e_u (sin), e_v (sin)
167  fours=f_cos; fouru=f_sin; fourv=f_sin; sparity = -1
168  END IF
169  r12 = sparity*hs_i/2
170 
171 !Calculate harmonics of electric field
172  CALL tomnsp_par(esubsijf, esubsmnf, fours)
173  CALL tomnsp_par(esubuijf, esubumnf, fouru)
174  CALL tomnsp_par(esubvijf, esubvmnf, fourv)
175 
176 !IMPOSE BOUNDARY CONDITION AT FIRST HALF-GRID POINT (SPH 04-28-16)
177  IF (startglobrow .EQ. 1) THEN
178 !These conditions are needed so that delta-W jogs (CheckForces) agree with forces at origin
179  esubsmnf(m2:,:,1) = 0 !f_v(1) wrong w/0 this
180  esubumnf(m0,:,1) = 0; esubumnf(m2:,:,1) = 0 !f_v(1) wrong w/0 this
181  esubvmnf(m1:,:,1) = 0
182 
183  IF (.NOT.l_natural) THEN
184  esubsmnh(m1,:,2) = (esubsmnf(m1,:,1) + esubsmnf(m1,:,2))/2
185  esubumnh(m1,:,2) = (esubumnf(m1,:,1) + esubumnf(m1,:,2))/2
186  djbsupsmnh(m1,:,1) = r12*esubsmnh(m1,:,2) - esubumnh(m1,:,2) !-> 0
187  !This constrains [esubs(2)*r12 - esubu(2)] = 0 at s=r12 (first half grid pt)
188  !needed to make djb^s ~ r12*djb^u there
189  esubsmnf(m1,:,1) = esubsmnf(m1,:,1) - djbsupsmnh(m1,:,1)/r12
190  esubumnf(m1,:,1) = esubumnf(m1,:,1) + djbsupsmnh(m1,:,1)
191  END IF
192  END IF
193 
194 !On exit, esubXh valid on [nsmin+1:nsmax]
195  CALL to_half_mesh(esubsmnf, esubsmnh)
196  CALL to_half_mesh(esubumnf, esubumnh)
197  CALL to_half_mesh(esubvmnf, esubvmnh)
198 
199 
200 !Gradients of full-grid esubu,v on half grid j-half=2,ns: note v(1) components used at j-half=2
201 !On exit, de_u,vds are valid on [nsmin+1:nsmax]
202 ! IF (nsmin .EQ. 1) esubumnf(m1,:,1) = 0 !Keep this: impacts Hessian symmetry!
203  CALL gradienthalf(de_uds, esubumnf)
204  CALL gradienthalf(de_vds, esubvmnf)
205 #undef _DUMP_BFIELD
206 !#define _DUMP_BFIELD
207 #if defined(_DUMP_BFIELD)
208  IF (nsmin.EQ.1 .AND. .NOT.l_compute_hessian .AND. &
209  any(abs(esubsmnf(m1,:,2)).GT.1.e-12_dp)) THEN
210  print *,'IPAR: ', ipar
211  print *,' n E_s(1) E_u(1)/r12 E_s(2) ' // &
212  'E_u(2)/r12 E_s(3) E_u(3)/r12'
213  DO n = -ntor, ntor
214  print 20, n, esubsmnf(m1,n,1), esubumnf(m1,n,1)/r12, &
215  esubsmnf(m1,n,2), esubumnf(m1,n,2)/r12, &
216  esubsmnf(m1,n,3), esubumnf(m1,n,3)/r12
217  END DO
218  20 FORMAT(i4,1p,6e12.4)
219  END IF
220 #endif
221 
222 !Compute contravariant components of -jacobian*curl(E) = r.h.s. of
223 !Faraday's equation used to evolve components of jacobh*BSUPX
224 !Need nmax=endglobrow+1 to compute perturbed currents on full mesh in siesta_forces
225  nmin=startglobrow
226  DO m = 0, mpol
227  mp = sparity*m ! -mp ~ d/du
228  DO n = -ntor, ntor
229  np = sparity*n*nfp ! -np ~ d/dv
230  djbsupsmnh(m,n,nmin:nmax) = mp*esubvmnh(m,n,nmin:nmax) &
231  - np*esubumnh(m,n,nmin:nmax)
232  djbsupumnh(m,n,nmin:nmax) =-np*esubsmnh(m,n,nmin:nmax) &
233  + de_vds(m,n,nmin:nmax)
234  djbsupvmnh(m,n,nmin:nmax) = mp*esubsmnh(m,n,nmin:nmax) &
235  - de_uds(m,n,nmin:nmax)
236  ENDDO
237  ENDDO
238 
239 !BOUNDARY CONDITIONS AT ORIGIN (used to compute bfields at origin in siesta_init, siesta_forces)
240  IF (nmin .EQ. 1) THEN
241  djbsupsmnh(:,:,1) = 0; djbsupsmnh(m1,:,1) = djbsupsmnh(m1,:,2)
242  djbsupumnh(:,:,1) = 0; djbsupumnh(m0:m1,:,1) = djbsupumnh(m0:m1,:,2)
243  djbsupvmnh(:,:,1) = 0; djbsupvmnh(m0,:,1) = djbsupvmnh(m0,:,2);
244  END IF
245 !
246 ! Calculate INCREMENT of magnetic field harmonics: use Euler scheme with DT given by VSUBMN advance equations.
247 ! Results computed here for HALF-GRID perturbations in MN space are used in the calculation of the force and
248 ! to advance the B's in UPDATE_STATE (in siesta_state module)
249 !
250  djbsupsmnh = delta_t*djbsupsmnh
251  djbsupumnh = delta_t*djbsupumnh
252  djbsupvmnh = delta_t*djbsupvmnh
253 
254  END SUBROUTINE faraday
