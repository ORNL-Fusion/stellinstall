218 ! Declare Arguments
219  REAL(rprec) :: pprofile_get_value
220  type(pprofile_class), INTENT (in) :: this
221  REAL(rprec), INTENT(in) :: s_arg
222 
223 ! local variables
224  REAL(rprec) :: s_use
225  REAL(rprec) :: s_01
226  LOGICAL :: l_01
227  INTEGER :: i, iflag
228  REAL (rprec) :: start_time
229 
230 ! Start of executable code
231  start_time = profiler_get_start_time()
232 
233 ! Define argument and logical variable for s between zero and one
234  s_01 = max(zero, min(one,s_arg))
235  l_01 = (s_arg .ge. zero) .and. (s_arg .le. one)
236 
237  SELECT CASE(this%p_type)
238 
239  CASE (pprofile_none_type)
240  pprofile_get_value = -1.e49_rprec
241 
242 ! v3fit uses an extra varible as an offset than vmec as a result the b array is
243 ! one index shorter than what two_power takes.
244  CASE (pprofile_two_power_type)
245  pprofile_get_value = this%b(0)
246  IF (l_01) THEN
247  pprofile_get_value = pprofile_get_value &
248  & + two_power(s_arg, this%b(1:iub_b))
249  ENDIF
250 
251 ! v3fit uses an extra varible as an offset than vmec as a result the b array is
252 ! one index shorter than what two_power_gs takes. As a result only 5 guassian
253 ! peaks maybe specified instead of the 6 vmec can use.
254  CASE (pprofile_two_power_gs_type)
255  pprofile_get_value = this%b(0)
256  IF (l_01) THEN
257  pprofile_get_value = pprofile_get_value &
258  & + two_power_gs(s_arg, &
259  & this%b(1:iub_b))
260  ENDIF
261 
262  CASE (pprofile_power_series_type)
263  pprofile_get_value = zero
264  DO i = iub_b, ilb_b, -1 ! all of b array, backwards
265  pprofile_get_value = pprofile_get_value*s_01 &
266  & + this%b(i)
267  END DO
268 
269  CASE (pprofile_cubic_spline_type)
270  s_use = min(this%as(this%maxSplineIndex - 1), &
271  & max(s_arg, this%as(1)))
272  CALL spline_cubic(s_use, pprofile_get_value, &
273  & this%as, this%af, &
274  & this%maxSplineIndex, iflag)
275  IF (iflag .ge. 0) THEN
276 
277  CALL profiler_set_stop_time('pprofile_get_value', &
278  & start_time)
279 
280  RETURN
281  ELSE IF (iflag .eq. -1) THEN
282  WRITE (*,*) 'ERROR: pprofile: outside value from ' // &
283  & 'spline_cubic'
284  ELSE IF(iflag .eq. -2) THEN
285  WRITE (*,*) 'ERROR: pprofile: decreasing s values ' // &
286  & 'in spline_cubic'
287  ELSE
288  WRITE (*,*) 'ERROR: pprofile: unknown error from ' // &
289  & 'spline_cubic'
290  ENDIF
291 ! Splines shuld not reach this point without triggering an error.
292  CALL exit(1)
293 
294  CASE (pprofile_akima_spline_type)
295  s_use = min(this%as(this%maxSplineIndex - 1), &
296  & max(s_arg,this%as(1)))
297  CALL spline_akima(s_use, pprofile_get_value, &
298  & this%as, this%af, &
299  & this%maxSplineIndex, iflag)
300  IF (iflag < 0) THEN
301  WRITE (*,*) 'ERROR: pprofile: bad value from ' // &
302  & 'spline_akima requested'
303  CALL exit(1)
304  ENDIF
305 
306  CASE (pprofile_line_segment_type)
307  CALL line_seg(s_arg, pprofile_get_value, this%as, this%af, &
308  & this%maxSplineIndex)
309 
310  END SELECT
311 
312  CALL profiler_set_stop_time('pprofile_get_value', start_time)
313
